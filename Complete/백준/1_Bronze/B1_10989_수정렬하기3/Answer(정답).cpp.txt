#include <bits/stdc++.h>
using namespace std;

int n, temp;
int numArr[10004];

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);
    cin >> n;

    // 입력 받기 : O(n)
    for (int i = 0; i < n; ++i)
    {
        cin >> temp;
        numArr[temp]++;
    }

    // 카운팅 정렬 출력 : O(10004 + n) = O(n)
    for (int i = 0; i < 10004; ++i) // 외부 루프 : O(10004)
    {
        // 내부 루프 : O(numArr[i]), 모든 i에 대한 총합 = O(n)
        for (int k = 0; k < numArr[i]; ++k)
        {
            cout << i << '\n';
        }
    }

    // 최종 시간 복잡도 = O(n + 10004 + n) = O(n)
    return 0;
}
