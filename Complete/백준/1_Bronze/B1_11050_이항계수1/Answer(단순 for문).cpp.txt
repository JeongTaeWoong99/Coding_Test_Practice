#include <bits/stdc++.h>
using namespace std;

int n, m;
int sum1 = 1, sum2 = 1;

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);
    
    cin >> n >> m;

    // 이항 계수 최적화: C(n, m) = [n × (n-1) × ... × (n-m+1)] / m!
    // 분자: n부터 (n-m+1)까지 곱셈 (m개)
    for (int i = n; i > n - m; --i)
        sum1 *= i;

    // 분모: m!
    for (int i = m; i > 1; --i)
        sum2 *= i;

    cout << sum1 / sum2;

    return 0;
}
