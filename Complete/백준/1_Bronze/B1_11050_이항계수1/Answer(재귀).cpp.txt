#include <bits/stdc++.h>
using namespace std;

int n, m;

// 팩토리얼 재귀 함수
int Factorial(int num)
{
    if (num == 1 || num == 0)
        return 1;

    return num * Factorial(num - 1);
}

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);
    
    cin >> n >> m;

    // 이항 계수 공식: C(n, m) = n! / (m! × (n-m)!)
    cout << Factorial(n) / (Factorial(m) * Factorial(n - m));

    return 0;
}
