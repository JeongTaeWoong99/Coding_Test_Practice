#include <bits/stdc++.h>
using namespace std;

int t;            // 테스트 케이스 개수
int k, n;         // k : 층, n : 호
int arr[20][20];  // DP 배열

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    cin >> t;

    // 0층 및 1호 초기화
    for (int i = 0; i <= 14; ++i)
    {
        arr[i][0] = 1;      // 모든 층의 1호는 1명
        arr[0][i] = i + 1;  // 0층의 i호는 i+1명
    }

    // DP 배열 채우기
    for (int i = 1; i <= 14; ++i)
    {
        for (int j = 1; j <= 14; ++j)
        {
            // k층 n호 = k층 (n-1)호 + (k-1)층 n호
            arr[i][j] = arr[i][j - 1] + arr[i - 1][j];
        }
    }

    // 테스트 케이스 처리
    for (int i = 0; i < t; ++i)
    {
        cin >> k >> n;
        n -= 1;  // 1호 → 인덱스 0으로 보정

        cout << arr[k][n] << '\n';
    }

    return 0;
}
