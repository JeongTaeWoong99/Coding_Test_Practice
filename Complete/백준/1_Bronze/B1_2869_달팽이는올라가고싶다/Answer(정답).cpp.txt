#include <bits/stdc++.h>
using namespace std;

int up, down, goal;
int diff;           // 하루 올라가는 높이 차이
int day;            // 필요 날짜

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(NULL);cout.tie(NULL);

    cin >> up >> down >> goal;

    // 하루 올라가는 높이 차이
    diff = up - down;

    // 마지막 업을 제외한 목표 높이
    int targetBeforeLastUp = goal - up;

    // 이 높이까지 도달하는데 필요한 날짜 (몫)
    day = targetBeforeLastUp / diff;

    // 나머지 확인으로 하루 더 필요한지 판단
    // 업 → 다운 → 업 → 다운 (targetBeforeLastUp 딱 맞음)
    // 딱 맞아떨어짐 = day일 동안 사이클 + 마지막 업 1일
    if (targetBeforeLastUp % diff == 0)
    {
        cout << day + 1 << "\n";
    }
    // 업 → 다운 → 업 → 다운 (targetBeforeLastUp 초과)
    // 나머지 있음 → 하루 더 필요 + 마지막 업 1일
    else
    {
        cout << day + 2 << "\n";
    }

    return 0;
}
