#include <bits/stdc++.h>
using namespace std;

const long long mode = 1234567891;  // 모듈러 값

char      c;
int       lenth;
long long ret;        // 해시 값 (long long으로 변경)
long long power = 1;  // 31^i를 저장할 변수 (1에서 부터 시작)
// 31^0 -> 31^1 -> 31^2 -> 31^i
// 1    -> 31   -> 961  -> ~~~

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    cin >> lenth;

    for (int i = 0; i < lenth; ++i)
    {
        cin >> c;

        // 알파벳 숫자로 변환 (+1 문제 보정)
        int changeNum = c - 'a' + 1;

        // 변환값 * 31^i (매 연산마다 mod로 오버플로우 방지)
        ret += (changeNum * power) % mode;   // 해시 값에 a_i * r^i 누적
        ret %= mode;                         // ret을 mode로 나눈 나머지로 업데이트

        // 다음 거듭제곱 계산 (31^(i+1) = 31^i * 31)
        power = (power * 31) % mode;
    }

    cout << ret << "\n";

    return 0;
}
