#include <bits/stdc++.h>
using namespace std;

int n;

int main()
{
    cin >> n;

    // i번째 줄 처리 (0부터 n-1까지)
    // 0 1 2 3 4 
    for (int i = 0; i < n; ++i)
    {
        // 앞 공백 출력 : n-i-1개
        // 4 3 2 1 0
        for (int j = 0; j < n - i - 1; ++j)
            cout << " ";

        // 별 출력 : 2*i+1개
        // 1 3 5 7 9
        for (int j = 0; j < i + i + 1; ++j)
            cout << "*";

        cout << '\n';
    }

    return 0;
}
