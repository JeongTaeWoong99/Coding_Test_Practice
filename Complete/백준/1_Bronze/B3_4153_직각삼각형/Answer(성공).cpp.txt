#include <bits/stdc++.h>
using namespace std;

int lines[3];
int val;

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    while (cin >> lines[0] >> lines[1] >> lines[2])
    {
        // 종료 케이스
        if (lines[0] == 0 && lines[1] == 0 && lines[2] == 0) break;

        // 가장 긴 라인 찾기(= 대각선)
        sort(lines, lines + 3);

        // c = 루트_a*a + b*b
        // 5 = 루트_3*3 + 4*4
        // 5 = 25 / 5
        val = (lines[0] * lines[0] + lines[1] * lines[1]);

        // 출력
        if (lines[2] == val / lines[2]) cout << "right" << '\n';
        else                            cout << "wrong" << '\n';
    }

    return 0;
}
