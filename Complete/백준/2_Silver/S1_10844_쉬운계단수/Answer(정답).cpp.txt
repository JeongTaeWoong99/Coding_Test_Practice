#include <bits/stdc++.h>
#define MOD 1000000000

using namespace std;

int n;
int dp[101][10];  // dp[i][j] : 길이가 i이고 마지막 자릿수가 j인 계단 수의 개수

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    cin >> n;

    // 1자리 계단 수 초기화 (0은 시작할 수 없으므로 1~9만)
    for(int i = 1; i <= 9; i++)
    {
        dp[1][i] = 1;
    }

    // 점화식 적용 : 길이 2부터 n까지
    for(int i = 2; i <= n; i++)
    {
        for(int j = 0; j <= 9; j++)
        {
            if(j == 0)
            {
                // 끝자리가 0이면 이전 자릿수는 1만 가능
                dp[i][0] = dp[i - 1][1];
            }
            else if(j == 9)
            {
                // 끝자리가 9이면 이전 자릿수는 8만 가능
                dp[i][9] = dp[i - 1][8];
            }
            else
            {
                // 끝자리가 j일 경우, 이전 자릿수는 j-1 또는 j+1
                dp[i][j] = dp[i - 1][j - 1] + dp[i - 1][j + 1];
            }

            // 오버플로 방지
            dp[i][j] %= MOD;
        }
    }

    // 길이 n인 계단 수의 총 개수 (끝자리 0~9 모두 합산)
    int result = 0;
    for(int j = 0; j <= 9; j++)
    {
        result = (result + dp[n][j]) % MOD;
    }

    cout << result << "\n";
    return 0;
}
