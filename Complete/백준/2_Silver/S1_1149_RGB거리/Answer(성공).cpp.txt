#include <bits/stdc++.h>
using namespace std;

int n;
int cost[1001][3]; // cost[i][j] : i번 집을 j색으로 칠하는 비용
int dp[1001][3];   // dp[i][j]   : i번 집을 j색으로 칠할 때까지의 최소 비용

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    // 입력 처리
    cin >> n;

    for (int i = 1; i <= n; ++i)
    {
        cin >> cost[i][0] >> cost[i][1] >> cost[i][2];
    }

    // DP 초기화 (1번 집)
    dp[1][0] = cost[1][0];
    dp[1][1] = cost[1][1];
    dp[1][2] = cost[1][2];

    // DP 점화식 : 이전 집과 다른 색 중 최솟값 선택
    for (int i = 2; i <= n; ++i)
    {
        dp[i][0] = cost[i][0] + min(dp[i-1][1], dp[i-1][2]);
        dp[i][1] = cost[i][1] + min(dp[i-1][0], dp[i-1][2]);
        dp[i][2] = cost[i][2] + min(dp[i-1][0], dp[i-1][1]);
    }

    // 결과 출력
    cout << min({ dp[n][0], dp[n][1], dp[n][2] }) << "\n";

    return 0;
}
