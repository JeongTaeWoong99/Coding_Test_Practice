#include <bits/stdc++.h>
using namespace std;

int n, m;
int dp[1025][1025]; // dp[i][j] : (1,1)~(i,j) 까지의 2D 누적합

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    cin >> n >> m;

    // 2D 누적합 구성
    for (int i = 1; i <= n; ++i)
    {
        for (int j = 1; j <= n; ++j)
        {
            int temp;
            cin >> temp;

            // 포함-배제 원리로 누적합 계산
            dp[i][j] = temp + dp[i - 1][j] + dp[i][j - 1] - dp[i - 1][j - 1];
        }
    }

    // 쿼리 처리 (각 쿼리 O(1))
    while (m--)
    {
        int x1, y1, x2, y2;
        cin >> x1 >> y1 >> x2 >> y2;

        // (x1,y1)~(x2,y2) 구간합 = 포함-배제 원리
        int sumCnt = dp[x2][y2] - dp[x1 - 1][y2] - dp[x2][y1 - 1] + dp[x1 - 1][y1 - 1];

        cout << sumCnt << '\n';
    }

    return 0;
}
