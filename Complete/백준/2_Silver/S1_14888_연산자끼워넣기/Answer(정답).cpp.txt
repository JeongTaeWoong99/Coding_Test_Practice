#include <bits/stdc++.h>
using namespace std;

// 전역 변수
int n;                     // 수의 개수
int arr[12];               // 주어진 수들
int maxRet = -1000000001;  // 최댓값
int minRet =  1000000001;  // 최솟값

// index : 현재 처리 중인 수의 인덱스
// cur : 현재까지의 계산 결과
// plus : 남은 덧셈 개수, minus : 남은 뺄셈 개수, mul : 남은 곱셈 개수, div : 남은 나눗셈 개수
void Back(int index, int cur, int plus, int minus, int mul, int div)
{
    // 모든 수를 사용한 경우
    if(index == n - 1)
    {
        maxRet = max(maxRet, cur); // 최댓값 갱신
        minRet = min(minRet, cur); // 최솟값 갱신
        return;
    }

    // 각 연산자를 사용 가능한 경우 재귀 호출
    if(plus)  Back(index + 1, cur + arr[index + 1], plus - 1, minus    , mul    , div);      // 덧셈
    if(minus) Back(index + 1, cur - arr[index + 1], plus    , minus - 1, mul    , div);      // 뺄셈
    if(mul)   Back(index + 1, cur * arr[index + 1], plus    , minus    , mul - 1, div);      // 곱셈
    if(div)   Back(index + 1, cur / arr[index + 1], plus    , minus    , mul    , div - 1);  // 나눗셈
}

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    // 입력 처리
    cin >> n;
    for(int i = 0; i < n; i++)
    {
        cin >> arr[i];
    }

    // 각 연산자의 개수
    int plusNum, minNum, mulNum, divNum;
    cin >> plusNum >> minNum >> mulNum >> divNum;

    // 백트래킹 시작
    // 시작값 = arr[0] = 처음 숫자
    Back(0, arr[0], plusNum, minNum, mulNum, divNum);

    // 결과 출력
    cout << maxRet << "\n" << minRet << "\n";

    return 0;
}
