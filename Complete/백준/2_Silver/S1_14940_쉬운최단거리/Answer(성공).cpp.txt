#include <bits/stdc++.h>
using namespace std;

const int dy[4] = {-1, 0, 1, 0};
const int dx[4] = {0, 1, 0, -1};

int n, m;              // n : 세로 크기, m : 가로 크기
int sY, sX;            // 시작 위치
int arr[1004][1004];   // 지도 (0:벽, 1:땅, 2:목표)
int dist[1004][1004];  // 목표지점까지의 거리 (-1로 초기화)

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    cin >> n >> m;

    // 입력 + 거리 배열 초기화
    for (int i = 0; i < n; i++)
    {
        for (int j = 0; j < m; j++)
        {
            cin >> arr[i][j];

            // 초기값 -1 (미방문)
            dist[i][j] = -1;

            // 목표지점(2) 위치 저장
            if (arr[i][j] == 2)
            {
                sY = i;
                sX = j;
            }
        }
    }

    // BFS 시작 (목표지점에서 출발)
    queue<pair<int, int>> q;
    dist[sY][sX] = 0;         // 목표지점 거리 = 0
    q.emplace(sY, sX);

    while (!q.empty())
    {
        int cy = q.front().first;
        int cx = q.front().second;
        q.pop();

        for (int i = 0; i < 4; i++)
        {
            int ny = cy + dy[i];
            int nx = cx + dx[i];

            // 범위 체크
            if (ny < 0 || ny >= n || nx < 0 || nx >= m) continue;

            // 벽이거나 이미 방문한 경우 스킵
            if (arr[ny][nx] == 0 || dist[ny][nx] != -1) continue;

            // 방문 처리 및 큐에 추가
            dist[ny][nx] = dist[cy][cx] + 1;
            q.emplace(ny, nx);
        }
    }

    // 결과 출력
    for (int i = 0; i < n; i++)
    {
        for (int j = 0; j < m; j++)
        {
            // 원래 갈 수 없는 땅(0)은 0 출력
            if (arr[i][j] == 0) cout << 0;
            else                cout << dist[i][j]; // 미도달 시 -1 그대로 출력

            if (j < m - 1)
                cout << " ";
        }
        cout << "\n";
    }

    return 0;
}
