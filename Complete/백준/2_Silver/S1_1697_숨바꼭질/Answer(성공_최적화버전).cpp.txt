#include <bits/stdc++.h>
using namespace std;

const int MAX = 200000;

int n, k;
int visited[MAX + 4];

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    cin >> n >> k;

    // N == K이면 이동 불필요
    if (n == k)
    {
        cout << 0;
        return 0;
    }

    // BFS 시작
    queue<int> bfsQ;
    visited[n] = 1;     // 방문 체크용(추후 출력에서 -1 해주기)
    bfsQ.push(n);       // n 시작위치 넣기

    while (!bfsQ.empty())
    {
        int now = bfsQ.front();
        bfsQ.pop();

        // 3가지 이동 : X-1, X+1, X*2
        for (int next : {now - 1, now + 1, now * 2})
        {
            // 범위 벗어나는지 체크
            if (next < 0 || next > MAX) continue;

            // 미방문인 경우(처음 방문!!!)
            if (!visited[next])
            {
                // 처음 방문이기 때문에, 지금 들어가는 값이 방문 최소값!!!
                visited[next] = visited[now] + 1;

                // BFS큐에 새로 넣기
                bfsQ.push(next);

                // 동생 위치 도달 시 즉시 출력
                if (next == k)
                {
                    cout << visited[k] - 1;
                    return 0;
                }
            }
        }
    }

    return 0;
}
