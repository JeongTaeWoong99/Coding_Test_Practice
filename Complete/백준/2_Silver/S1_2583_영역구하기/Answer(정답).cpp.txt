#include <bits/stdc++.h>
using namespace std;

const int dY[4] = {0, -1, 0, 1};
const int dX[4] = {1, 0, -1, 0};

int    N, M, K, ret;
int    x_1, x_2, y_1, y_2;
int    arr[104][104];
int    vit[104][104];
vector<int> v;  // 영역 넓이 저장 벡터

// BFS : 연결된 영역의 넓이 계산
int BFS(int y, int x)
{
    int areaCount = 1;

    vit[y][x] = 1;  // 방문 처리
    queue<pair<int, int>> q;
    q.emplace(y, x);

    while(!q.empty())
    {
        int y = q.front().first;
        int x = q.front().second;
        q.pop();  // 큐에서 제거

        for (int i = 0; i < 4; i++)
        {
            int newY = y + dY[i];
            int newX = x + dX[i];

            // 범위 체크
            if (newY < 0 || newY >= N || newX < 0 || newX >= M) continue;

            // arr 1 자리는 패스 (직사각형 영역)
            if (arr[newY][newX] == 1) continue;

            // 방문 패스
            if (vit[newY][newX] == 1) continue;

            areaCount++;

            vit[newY][newX] = 1;
            q.emplace(newY, newX);
        }
    }

    return areaCount;
}

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    cin >> N >> M >> K;

    // 직사각형 영역 표시
    for(int i = 0; i < K; i++)
    {
        cin >> x_1 >> y_1 >> x_2 >> y_2;

        for(int x = x_1; x < x_2; x++)
        {
            for(int y = y_1; y < y_2; y++)
            {
                arr[y][x] = 1;
            }
        }
    }

    // 모든 구역 탐색 및 기록
    for(int i = 0; i < N; i++)
    {
        for(int j = 0; j < M; j++)
        {
            // arr 자리가 0 && vit가 0이면 구역 탐색
            if (arr[i][j] == 0 && vit[i][j] == 0)
            {
                ret++;                     // 영역 개수 증가
                v.emplace_back(BFS(i, j)); // 영역 넓이 저장
            }
        }
    }

    // 넓이 오름차순 정렬
    sort(v.begin(), v.end());

    // 결과 출력
    cout << ret << "\n";
    for (int num : v)
        cout << num << " ";

    return 0;
}
