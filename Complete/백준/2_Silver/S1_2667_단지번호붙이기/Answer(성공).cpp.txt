#include <bits/stdc++.h>
using namespace std;

const int dy[4] = {0, 1,  0, -1};
const int dx[4] = {1, 0, -1,  0};

int         n;
int         arr[30][30];  // 1 = 집, 0 = 빈집
bool        vit[30][30];  // 방문 체크
vector<int> areaV;        // 그룹 넓이 순서대로 저장(+ 그룹 수 체크도 가능)

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    // n x n 크기
    cin >> n;

    // 배열 입력
    for (int i = 0; i < n; ++i)
    {
        string s;
        cin >> s;

        for (int j = 0; j < n; ++j)
        {
            arr[i][j] = s[j] - '0';
        }
    }

    // n x n 배열의 모든 지점을 탐색
    for (int i = 0; i < n; ++i)
    {
        for (int j = 0; j < n; ++j)
        {
            // 집이 있고 && 방문을 하지 않았으면
            // => 해당 지점에서 탐색 시작
            if (arr[i][j] == 1 && vit[i][j] == false)
            {
                // BFS 탐색
                queue<pair<int,int>> bfsQ;
                bfsQ.emplace(i,j);
                vit[i][j] = true;

                // 해당 구역 넓이 저장 후, areaV에 순서대로 저장
                int areaCnt = 1;

                // 구역 체크 시작
                while (!bfsQ.empty())
                {
                    // 맨 앞 꺼내오기 및 제거
                    int curY = bfsQ.front().first;
                    int curX = bfsQ.front().second;
                    bfsQ.pop();

                    // 4방향 탐색
                    for (int k = 0; k < 4; ++k)
                    {
                        int newY = curY + dy[k];
                        int newX = curX + dx[k];

                        // 범위 체크
                        if (newX < 0 || newX >= n || newY < 0 || newY >= n) continue;

                        // 집인지 체크
                        if (arr[newY][newX] == 0) continue;

                        // 방문 했는지 체크
                        if (vit[newY][newX] == true) continue;

                        bfsQ.emplace(newY,newX);
                        vit[newY][newX] = true;

                        areaCnt++;
                    }
                }

                // 에리어 넓이 순서대로 저장
                areaV.push_back(areaCnt);
            }
        }
    }

    // 전체 그룹 수
    cout << (int)areaV.size() << '\n';

    // 각 그룹의 크기(오름차순 후 출력)
    sort(areaV.begin(), areaV.end());
    for (int curArea : areaV)
        cout << curArea << '\n';

    return 0;
}
