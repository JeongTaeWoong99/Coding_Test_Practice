// 각 위치에서 PN 패턴과 substr 비교
// O(N*M) : N ≤ 100, M ≤ 10,000 서브태스크만 통과

#include <bits/stdc++.h>
using namespace std;

int    n, m, cnt;           // PN 길이, 스트링 길이, 결과값
string s, checkS = "IOI";   // 입력 스트링, 체크 스트링 (PN)

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    cin >> n >> m >> s;

    // PN 완성 : "IOI" + "OI" * (n-1)
    for (int i = 1; i < n; i++)
        checkS += "OI";

    // 각 시작 위치에서 substr로 PN과 비교
    for (int i = 0; i <= m - (int)checkS.size(); i++)
    {
        if (s.substr(i, checkS.size()) == checkS)
            cnt++;
    }

    // 결과 출력
    cout << cnt;

    return 0;
}
