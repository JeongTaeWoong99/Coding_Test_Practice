#include <bits/stdc++.h>
using namespace std;

const int INF = 2008;  // 무한대 값 (문제 조건: 1 ≤ Ai ≤ 1000)

int n;                 // 수열의 크기
int num;               // 입력받을 숫자
int lis[1004];         // LIS를 저장하는 배열
int len;               // 현재 LIS의 길이

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    // LIS 배열 초기화 (범위보다 크게)
    fill_n(lis, 1004, INF);

    // 입력 처리
    cin >> n;

    // LIS 구하기 (이분 탐색)
    for (int i = 0; i < n; i++)
    {
        cin >> num;

        // lower_bound로 num이 들어갈 위치 찾기
        int pos = lower_bound(lis, lis + len, num) - lis;

        // 새로운 값이면 길이 증가
        if (lis[pos] == INF) len++;

        // LIS 배열 갱신
        lis[pos] = num;
    }

    // 결과 출력
    cout << len << '\n';

    return 0;
}
