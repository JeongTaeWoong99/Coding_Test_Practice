#include <bits/stdc++.h>
using namespace std;

int         n;
vector<int> adj[100001];     // 인접 리스트
int         parent[100001];  // 각 노드의 부모
bool        visited[100001]; // 방문 여부

void BFS()
{
    queue<int> q;
    q.push(1);
    visited[1] = true;

    while(!q.empty())
    {
        int cur = q.front();
        q.pop();

        for(int next : adj[cur])
        {
            if(visited[next]) continue;

            visited[next] = true;   // 방문 처리
            parent[next]  = cur;    // cur가 next의 부모
            q.push(next);           // 새로 넣기
        }
    }
}

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    cin >> n;

    for(int i = 0; i < n - 1; i++)
    {
        // 연결 입력
        int u, v;
        cin >> u >> v;

        // 양방향 연결
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    BFS();

    for(int i = 2; i <= n; i++)
    {
        cout << parent[i] << "\n";
    }

    return 0;
}
