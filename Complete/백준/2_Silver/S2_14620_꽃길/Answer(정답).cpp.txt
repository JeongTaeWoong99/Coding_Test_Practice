#include <bits/stdc++.h>
using namespace std;

const int dy[4] = {-1, 1, 0, 0};  // 상하좌우 y 방향 이동
const int dx[4] = {0, 0, -1, 1};  // 상하좌우 x 방향 이동

// 전역 변수 선언
int N;                      // 화단의 크기 (N x N)
int ret = 987654321;        // 최소 비용 (최댓값으로 초기화. 초기화 값에서 점점 작아짐)
int visited[20][20];        // 방문 체크 배열 (꽃이 심어진 위치 표시)
int arr[20][20];            // 화단의 각 위치별 가격

// 꽃을 심고 비용을 계산하는 함수
// y : 꽃의 중심 y 좌표, x : 꽃의 중심 x 좌표
// 반환값 : 꽃을 심는데 필요한 총 비용 (중심 + 4개 꽃잎)
int SetFlower(int y, int x)
{
    int sum = 0; // 5개 비용 합

    visited[y][x] = 1; // 중심 방문 처리
    sum += arr[y][x];  // 중심 비용 더하기

    // 상하좌우 4개 꽃잎 심기
    for(int i = 0; i < 4; i++)
    {
        int ny = y + dy[i];
        int nx = x + dx[i];

        visited[ny][nx]  = 1;           // 꽃잎 위치 방문 처리
        sum             += arr[ny][nx]; // 꽃잎 비용 더하기
    }

    return sum;  // 총 비용 반환 (5개 칸의 합)
}

// 해당 위치에 꽃을 심을 수 있는지 확인하는 함수
// y : 확인할 y 좌표, x : 확인할 x 좌표
// 반환값 : true(심을 수 있음) / false(심을 수 없음)
bool Check(int y, int x)
{
    // 1. 중심이 이미 사용된 위치인지 확인
    if(visited[y][x]) return 0;

    // 2. 상하좌우 4개 꽃잎 위치 확인
    for(int i = 0; i < 4; i++)
    {
        int ny = y + dy[i];
        int nx = x + dx[i];

        // 2-1. 화단 밖으로 나가는지 확인
        // 2-2. 이미 다른 꽃이 심어진 위치인지 확인
        if(ny < 0 || ny >= N || nx < 0 || nx >= N || visited[ny][nx])
        {
            return 0;  // 하나라도 조건 위반 시 심을 수 없음
        }
    }

    return 1;  // 모든 조건 만족 시 심을 수 있음
}

// 백트래킹을 위해 심었던 꽃을 제거하는 함수 (방문 체크 해제)
// y : 꽃의 중심 y 좌표, x : 꽃의 중심 x 좌표
void EraseFlower(int y, int x)
{
    visited[y][x] = 0;  // 중심 방문 해제

    // 상하좌우 4개 꽃잎 방문 해제
    for(int i = 0; i < 4; i++)
    {
        int ny = y + dy[i];
        int nx = x + dx[i];
        visited[ny][nx] = 0;  // 꽃잎 위치 방문 해제
    }
}

// 백트래킹으로 3개의 꽃을 심는 모든 경우의 수를 탐색하는 함수
// cnt : 현재까지 심은 꽃의 개수
// hap : 현재까지의 총 비용
void Flower(int cnt, int hap)
{
    // 기저 조건 : 3개의 꽃을 모두 심었을 때
    if(cnt == 3)
    {
        ret = min(ret, hap);  // 최소 비용 갱신
        return;
    }

    // 화단의 모든 위치를 탐색
    for(int i = 0; i < N; i++)
    {
        for(int j = 0; j < N; j++)
        {
            // 해당 위치에 꽃을 심을 수 있는지 확인
            if(Check(i, j))
            {
                // 꽃을 심고 재귀 호출 (비용 누적)
                Flower(cnt + 1, hap + SetFlower(i, j));

                // 백트래킹: 심었던 꽃 제거 (다른 경우의 수 탐색을 위해)
                EraseFlower(i, j);
            }
        }
    }
}

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    // 입력 : 화단 크기
    cin >> N;

    // 입력 : 화단의 각 위치별 가격
    for(int i = 0; i < N; i++)
    {
        for(int j = 0; j < N; j++)
        {
            cin >> arr[i][j];
        }
    }

    // 백트래킹으로 3개의 꽃을 심는 모든 경우의 수 탐색
    Flower(0, 0);

    // 출력: 최소 비용
    cout << ret << "\n";

    return 0;
}
