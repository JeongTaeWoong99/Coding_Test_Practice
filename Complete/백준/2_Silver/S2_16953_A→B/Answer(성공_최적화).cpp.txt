\xEF\xBB\xBF#include <bits/stdc++.h>
using namespace std;

long long                a, b;
unordered_set<long long> visited;  // 동일한 값 재방문 방지

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    cin >> a >> b;

    // BFS : a에서 시작해 최소 연산 횟수로 b에 도달하는 경로 탐색
    // pair : {현재 값, 지금까지 연산 횟수}
    queue<pair<long long, int>> bfsQ;
    bfsQ.emplace(a, 0);
    visited.insert(a);

    while(!bfsQ.empty())
    {
        auto [cur, cnt] = bfsQ.front();
        bfsQ.pop();

        // b에 도달 → 최솟값 + 1 출력
        if(cur == b)
        {
            cout << cnt + 1 << "\n";
            return 0;
        }

        long long nextOne = cur * 2;       // [연산 1] ×2
        long long nextTwo = cur * 10 + 1;  // [연산 2] 끝에 1 추가

        // next1이 b 이하이고 미방문이면 큐에 추가
        if(nextOne <= b && !visited.count(nextOne))
        {
            visited.insert(nextOne);
            bfsQ.emplace(nextOne, cnt + 1);
        }

        // next2가 b 이하이고 미방문이면 큐에 추가
        if(nextTwo <= b && !visited.count(nextTwo))
        {
            visited.insert(nextTwo);
            bfsQ.emplace(nextTwo, cnt + 1);
        }
    }

    // 큐가 소진될 때까지 b에 도달 못하면 불가능
    cout << -1 << "\n";

    return 0;
}
