#include <bits/stdc++.h>
using namespace std;

// 전역 변수
int n;              // 수열의 크기
int newNum;         // 입력받을 현재 수
int curSum = 0;     // 현재까지의 연속합
int maxRet = -1001; // 최대 연속합 (최솟값으로 초기화)

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    cin >> n;
    for(int i = 0; i < n; i++)
    {
        cin >> newNum;

        curSum += newNum;                       // 현재 수를 연속합에 추가
        maxRet  = max(maxRet, curSum); // 최댓값 갱신

        // 음수가 되면 초기화 (Kadane's Algorithm)
        if(curSum < 0)
            curSum = 0;
    }

    // 결과 출력
    cout << maxRet << "
";
    
    return 0;
}
