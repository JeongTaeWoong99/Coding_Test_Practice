#include <bits/stdc++.h>
using namespace std;

const int OFFSET  = 4000;   // 음수 보정값 (-4000 ~ 4000 → 0 ~ 8000)
const int MAX_VAL = 8001;   // 배열 크기

int n, temp;
int sum;            // 산술평균
vector<int> v;      // 중앙값 및 범위
int freq[MAX_VAL];  // 최빈값
vector<int> freqV;  // 여러개 최빈값 저장용

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    cin >> n;

    // 미리 할당 (재할당 방지)
    v.reserve(n);
    freqV.reserve(n);

    for (int i = 0; i < n; ++i)
    {
        cin >> temp;

        // 모든 숫자 합 (산술평균 용)
        sum += temp;

        // 모든 숫자 벡터 저장 (중앙값 용)
        v.emplace_back(temp);

        // 숫자 사용 빈도 저장 (- 값도 있기 때문에 OFFSET 보정해서 저장)
        freq[temp + OFFSET]++;
    }

    // 산술평균 (반올림 -> double형으로 소수까지 계산하고, round로 반올림 후, int로 절삭)
    cout << (int)round((double)sum / n) << '\n';

    // 중앙값 (정렬 후 중간 인덱스)
    sort(v.begin(), v.end());
    cout << v[n / 2] << '\n';

    // 최빈값
    int maxUseNum = 0;
    for (int i = 0; i < MAX_VAL; ++i)
    {
        // 사용 x 패스
        if (freq[i] == 0) continue;

        // 더 사용 횟수가 큼 => 벡터 교체
        if (freq[i] > maxUseNum)
        {
            maxUseNum = freq[i];            // 최대 사용 횟수 변경
            freqV.clear();                  // 벡터 비우기
            freqV.emplace_back(i - OFFSET); // 벡터에 값 넣기 (값 보정)
        }
        // 사용 횟수가 동일함 => 벡터 추가
        else if (maxUseNum == freq[i])
        {
            freqV.emplace_back(i - OFFSET); // 벡터에 값 넣기
        }
    }

    // 0 ~ MAX_VAL을 차례로 확인하기 때문에, 이미 정렬 되어 있음.
    if   (freqV.size() == 1) cout << freqV[0] << '\n'; // 사이즈가 1인 경우     => 처음 출력
    else                     cout << freqV[1] << '\n'; // 사이즈가 2 이상인 경우 => 두번째 출력

    // 범위
    cout << v.back() - v.front();

    return 0;
}
