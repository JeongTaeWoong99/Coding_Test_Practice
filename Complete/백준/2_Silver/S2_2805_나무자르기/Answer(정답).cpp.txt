#include <bits/stdc++.h>
using namespace std;

int       n;                // 나무의 수
long long m;                // 필요한 나무 길이 (최대 20억 → long long)
long long ar[1000004];      // 나무 높이 배열
long long maxHeight = 0;    // 가장 높은 나무

// mid 높이로 잘랐을 때, 얻는 나무의 총 길이
long long GetWood(long long mid)
{
    long long sum = 0;

    for (int i = 0; i < n; ++i)
    {
        if (ar[i] > mid)
            sum += ar[i] - mid;  // mid보다 높은 부분만 획득
    }

    return sum;
}

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    cin >> n >> m;

    for (int i = 0; i < n; ++i)
    {
        cin >> ar[i];
        maxHeight = max(maxHeight, ar[i]);  // 최대 높이 갱신
    }

    // 이분 탐색: M미터 이상 얻을 수 있는 H의 최댓값
    long long lo = 0;
    long long hi = maxHeight;
    long long ret = 0;

    while (lo <= hi)
    {
        long long mid = (lo + hi) / 2;
        long long wood = GetWood(mid);

        // M 이상 얻을 수 있으면 → 더 높이 잘라도 될 수 있음 (H 최댓값 찾기)
        if (wood >= m)
        {
            ret = mid;    // 현재 높이 저장
            lo = mid + 1; // 더 높은 H 탐색
        }
        // M 미만이면 → 더 낮게 잘라야 함
        else
        {
            hi = mid - 1;
        }
    }

    cout << ret;

    return 0;
}
