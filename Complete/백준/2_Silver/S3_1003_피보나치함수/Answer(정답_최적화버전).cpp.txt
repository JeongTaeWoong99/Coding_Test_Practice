// 백준 1003번 - 피보나치 함수
// 다이나믹 프로그래밍 - 전처리 방식

#include <bits/stdc++.h>
using namespace std;

int t, n;
int dp[41] = {0, 1, };  // 피보나치 수열

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    // ★ 전처리 : 피보나치 수열을 미리 계산 (한 번만!)
    for(int i = 2; i <= 40; i++)
        dp[i] = dp[i - 1] + dp[i - 2];

    cin >> t;
    while(t--)
    {
        cin >> n;

        // 인덱스 범위 예외 : fib(-1)은 없음
        if(n == 0)
            cout << "1 0\n";
        // 0 호출 횟수 = fib(n-1), 1 호출 횟수 = fib(n)
        else
            cout << dp[n - 1] << " " << dp[n] << "\n";
    }

    return 0;
}
