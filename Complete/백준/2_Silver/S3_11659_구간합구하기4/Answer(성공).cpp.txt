#include <bits/stdc++.h>
using namespace std;

int n, m, temp;
int arr[100004]; // 누적합 배열

int main()
{
    ios_base::sync_with_stdio(0),cin.tie(0),cout.tie(0);
    
    cin >> n >> m;

    // 누적합을 위해, 1번부터 시작
    // 범위 구간 숫자도 1부터 시작
    for (int i = 1; i <= n; ++i)
    {
        cin >> temp;

        arr[i] = temp + arr[i - 1]; // 현재 자리 누적값 = 자리값 + 이전 누적값
    }

    // 구하는 값 = b까지의 누적합 - a 이전 자리까지의 누적값
    int a, b;
    while (m--)
    {
        cin >> a >> b;
        
        cout << arr[b] - arr[a - 1]  << '\n';
    }

    return 0;
}
