#include <bits/stdc++.h>
using namespace std;

int n;
int dp[1004];

int main()
{
    ios_base::sync_with_stdio(0),cin.tie(0),cout.tie(0);

    cin >> n;

    dp[0] = 0;
    dp[1] = 1;
    dp[2] = 2;

    for (int i = 3; i <= n; ++i)
    {
        dp[i] = (dp[i - 1] + dp[i - 2]) % 10007; // 매 단계 모듈러 (오버플로우 방지)
    }

    cout << dp[n];
    
    return 0;
}
