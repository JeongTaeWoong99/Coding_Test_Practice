#include <bits/stdc++.h>
using namespace std;

int    n, m, cnt, temp;
map<int, int> _map;

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    cin >> n >> m;

    // A 집합 입력 및 카운트
    for(int i = 0; i < n; i++)
    {
        cin >> temp;
        _map[temp]++;
    }

    // B 집합 입력 및 카운트
    for(int i = 0; i < m; i++)
    {
        cin >> temp;
        _map[temp]++;
    }

    // 한 번만 나온 원소 = 대칭 차집합
    for(auto it : _map)
    {
        if(it.second == 1)
            cnt++;
    }

    cout << cnt << "\n";

    return 0;
}
