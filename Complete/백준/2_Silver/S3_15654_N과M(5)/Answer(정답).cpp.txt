#include <bits/stdc++.h>
using namespace std;

int  n, m;
int  nums[9];    // 입력받은 N개의 자연수 (정렬 후 사용)
int  arr[9];     // 현재 선택된 수열
bool visited[9]; // 방문 여부

// depth : 현재 선택한 수의 개수
void DFS(int depth)
{
    if(depth == m)
    {
        for(int i = 0; i < m; i++)
        {
            cout << arr[i];

            if(i == m - 1) cout << "\n"; // 마지막 원소 → 줄바꿈
            else           cout << " ";  // 중간 원소  → 공백
        }
        return;
    }

    for(int i = 0; i < n; i++)
    {
        if(!visited[i])
        {
            visited[i] = true;
            arr[depth]  = nums[i];

            DFS(depth + 1);

            visited[i] = false; // 백트래킹
        }
    }
}

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    cin >> n >> m;
    
    for(int i = 0; i < n; i++) 
        cin >> nums[i];

    sort(nums, nums + n); // 사전순 출력을 위해 정렬

    DFS(0);

    return 0;
}
