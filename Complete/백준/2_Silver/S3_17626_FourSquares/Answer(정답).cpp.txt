#include <bits/stdc++.h>
using namespace std;

int n;
int dp[50001];  // dp[i] : i를 제곱수 합으로 표현하는 최소 개수

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    cin >> n;

    // 최악의 경우로 초기화 (전부 1로 표현 : 1+1+1+...+1)
    for (int i = 1; i <= n; i++)
        dp[i] = i;

    // DP 점화식 적용
    for (int i = 1; i <= n; i++)
    {
        // 제곱은 j * j <= i를 넘어갈 수 없음
        // j²를 사용하면 1개 추가
        for (int j = 1; j * j <= i; j++)
        {
            dp[i] = min(dp[i], dp[i - j * j] + 1);
        }
    }

    cout << dp[n];

    return 0;
}
