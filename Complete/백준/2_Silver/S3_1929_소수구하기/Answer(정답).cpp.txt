#include <bits/stdc++.h>
using namespace std;

bool isPrime[1000001];  // true : 소수, false : 소수 아님
int  m, n;

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    cin >> m >> n;

    // 일단 모두 소수로 가정 (0, 1은 소수 아님)
    fill_n(&isPrime[0], 1000001, true);
    isPrime[0] = isPrime[1] = false;

    // 에라토스테네스의 체
    for (int i = 2; i * i <= n; i++)
    {
        // 이미 소수가 아니면 스킵
        if (!isPrime[i]) continue;

        // i*i부터 i의 배수 제거
        for (int j = i * i; j <= n; j += i)
            isPrime[j] = false;
    }

    // M 이상 N 이하 소수 출력
    for (int i = m; i <= n; i++)
        if (isPrime[i])
            cout << i << '\n';

    return 0;
}
