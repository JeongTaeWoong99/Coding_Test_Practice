#include <bits/stdc++.h>
using namespace std;

int n;
int a[304];   // 각 계단의 점수
int dp[304];  // dp[i] : i번째 계단까지 올라왔을 때의 최대 점수

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    cin >> n;
    for (int i = 1; i <= n; i++)
        cin >> a[i];

    // 초기값 설정 (Base case)
    dp[1] = a[1];                          // 첫 번째 계단
    dp[2] = a[1] + a[2];                   // 첫 두 계단 (연속 2개는 OK)
    dp[3] = max(a[1] + a[3], a[2] + a[3]); // 1-3 or 2-3

    // 점화식 : dp[i] = max(2칸 점프, 1칸+1칸)
    // - 2칸 점프 : dp[i-2] + a[i]           (i-1 안 밟음)
    // - 1칸+1칸 : dp[i-3] + a[i-1] + a[i]   (i-2 안 밟음)
    // → 둘 다 중간에 하나를 건너뛰므로 3연속 불가능!
    for (int i = 4; i <= n; i++)
    {
        dp[i] = max(
            dp[i - 2] + a[i],
            dp[i - 3] + a[i - 1] + a[i]
        );
    }

    cout << dp[n];

    return 0;
}
