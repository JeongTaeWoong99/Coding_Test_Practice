#include <bits/stdc++.h>
using namespace std;

int n, a;

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(NULL);cout.tie(NULL);

    cin >> n;

    for(int i = 0; i < n; i++)
    {
        cin >> a; // N! 할 숫자

        int ret5 = 0; // 5의 개수

        // a를 5로 나눠서, 몇개 나오는지 체크
        // 5의 배수로만 체크하기 위해, j *= 5
        for(int j = 5; j <= a; j *= 5)
        {
            ret5 += a / j;
        }

        cout << ret5 << "\n";
    }

    return 0;
}
