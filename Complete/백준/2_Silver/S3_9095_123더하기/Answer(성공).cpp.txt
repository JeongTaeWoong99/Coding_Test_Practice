#include <bits/stdc++.h>
using namespace std;

int n, temp;
int dp[14];

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    cin >> n;

    // 베이스 케이스
    dp[0] = 1;  // 아무것도 안 더해서 0을 만드는 경우 = 1가지
    dp[1] = 1;  // 1
    dp[2] = 2;  // 1+1, 2
    dp[3] = 4;  // 1+1+1, 1+2, 2+1, 3

    // dp 채우기(4부터 ~ 10까지)
    // i자리 값은 기록된 -1 -2 -3 자리를 더한 값
    for (int i = 4; i <= 10; ++i)
    {
        dp[i] = dp[i-1] + dp[i-2] + dp[i-3];
    }

    // 출력
    while (n--)
    {
        cin >> temp;

        cout << dp[temp] << '\n';
    }

    return 0;
}
