#include <bits/stdc++.h>
using namespace std;

int       n;
long long dp[104];  // int 오버플로우 주의! N ≤ 100이면 long long 필수

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    cin >> n;

    // 베이스 케이스
    dp[0] = 1;
    dp[1] = 1;
    dp[2] = 1;

    // 바텀업 DP(i자리 값은 -3뒤 -2뒤 자리의 합)
    for (int i = 3; i <= 100; ++i)
    {
        dp[i] = dp[i - 3] + dp[i - 2];
    }

    // 출력
    int temp;
    while (n--)
    {
        cin >> temp;

        cout << dp[temp - 1] << '\n';   // 위치 보정
    }

    return 0;
}
