#include <bits/stdc++.h>
using namespace std;

const int OFFSET = 10000000;  // 음수 처리를 위한 오프셋

int n, m, temp;
int cnt[20000001];  // -10,000,000 ~ 10,000,000 매핑

int main()
{
    ios_base::sync_with_stdio(0),cin.tie(0),cout.tie(0);
    
    cin >> n;
    for (int i = 0; i < n; ++i)
    {
        cin >> temp;
        cnt[temp + OFFSET]++;  // 음수 처리 (오프셋 더하기)
    }
    
    cin >> m;
    for (int i = 0; i < m; ++i)
    {
        cin >> temp;
        cout << cnt[temp + OFFSET] << " ";
    }
    
    return 0;
}
