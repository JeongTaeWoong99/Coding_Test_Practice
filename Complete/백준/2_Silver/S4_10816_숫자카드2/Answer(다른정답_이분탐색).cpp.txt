#include <bits/stdc++.h>
using namespace std;

int        n, m, temp;
vector<int> cards;

int main()
{
    ios_base::sync_with_stdio(0),cin.tie(0),cout.tie(0);
    
    cin >> n;
    cards.reserve(n);  // 메모리 재할당 방지
    
    for (int i = 0; i < n; ++i)
    {
        cin >> temp;
        cards.push_back(temp);
    }
    
    // 이분 탐색을 위한 정렬
    sort(cards.begin(), cards.end());
    
    cin >> m;
    for (int i = 0; i < m; ++i)
    {
        cin >> temp;
        
        // equal_range : lower_bound + upper_bound 한 번에
        auto range = equal_range(cards.begin(), cards.end(), temp);
        cout << range.second - range.first << " ";
    }
    
    return 0;
}
