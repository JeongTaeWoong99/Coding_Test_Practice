#include <bits/stdc++.h>
using namespace std;

int                    n, m, temp;
unordered_map<int,int> unMap;

int main()
{
    ios_base::sync_with_stdio(0),cin.tie(0),cout.tie(0);
    
    cin >> n;
    unMap.reserve(n);  // 해시 테이블 버킷 미리 할당 (리해싱 방지)
    
    for (int i = 0; i < n; ++i)
    {
        cin >> temp;
        unMap[temp]++;  // 해당 숫자 카운트
    }
    
    cin >> m;
    for (int i = 0; i < m; ++i)
    {
        cin >> temp;
        
        // find()로 불필요한 키 생성 방지
        auto it = unMap.find(temp);
        cout << (it != unMap.end() ? it->second : 0) << " ";
    }
    
    return 0;
}
