#include <bits/stdc++.h>
using namespace std;

// n   : 명령의 수
// x   : push 명령의 정수
// s   : 명령어 문자열
// stk : 정수를 저장하는 스택
int        n, x;
string     s;
stack<int> stk;

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    cin >> n;

    for(int i = 0; i < n; i++)
    {
        cin >> s;

        if(s == "push")
        {
            cin >> x;
            stk.push(x);
        }
        else if(s == "pop")
        {
            if(stk.empty())
                cout << -1 << "\n";
            else
            {
                cout << stk.top() << "\n";
                stk.pop();
            }
        }
        else if(s == "size")
        {
            cout << stk.size() << "\n";
        }
        else if(s == "empty")
        {
            cout << (stk.empty() ? 1 : 0) << "\n";
        }
        else if(s == "top")
        {
            if(stk.empty())
                cout << -1 << "\n";
            else
                cout << stk.top() << "\n";
        }
    }

    return 0;
}
