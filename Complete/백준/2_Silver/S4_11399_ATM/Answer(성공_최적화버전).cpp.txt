// 백준 11399번 - ATM (Silver 4)
// 알고리즘: 그리디 + 정렬
// 최적화: 2중 for문 → 단일 for문 (가중치 곱셈)
// 시간복잡도: O(N log N)

#include <bits/stdc++.h>
using namespace std;

int n, ans;
int arr[1000];  // 고정 배열 (N ≤ 1000)

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    cin >> n;

    for (int i = 0; i < n; ++i)
        cin >> arr[i];

    sort(arr, arr + n);  // 오름차순 정렬

    // 단일 for문: P[i]는 (n-i)번 더해짐
    for (int i = 0; i < n; ++i)
        ans += arr[i] * (n - i);

    cout << ans;

    return 0;
}
