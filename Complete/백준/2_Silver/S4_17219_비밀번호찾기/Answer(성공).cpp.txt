#include <bits/stdc++.h>
using namespace std;

// 전역 변수
int n, m;                          // n : 저장된 사이트 수, m : 찾을 사이트 수
unordered_map<string, string> pw;  // pw : 사이트 주소 → 비밀번호 매핑

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    // 입력 처리
    cin >> n >> m;

    // N개의 사이트 주소와 비밀번호 저장
    string site, password;
    for(int i = 0; i < n; i++)
    {
        cin >> site >> password;
        pw[site] = password;            // 해시맵에 저장
    }

    // M개의 쿼리 처리
    string query;
    for(int i = 0; i < m; i++)
    {
        cin >> query;
        cout << pw[query] << "\n";      // 해당 사이트의 비밀번호 출력
    }

    return 0;
}
