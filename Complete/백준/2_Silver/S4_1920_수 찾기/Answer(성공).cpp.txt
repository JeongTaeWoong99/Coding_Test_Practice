#include <bits/stdc++.h>
using namespace std;

int                n, temp;
unordered_set<int> st;

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);
    
    cin >> n;
    for (int i = 0; i < n; ++i)
    {
        cin >> temp;
        st.emplace(temp);
    }

    cin >> n;
    for (int i = 0; i < n; ++i)
    {
        cin >> temp;
        if (st.find(temp) != st.end()) cout << "1" << '\n';
        else                           cout << "0" << '\n';
    }

    return 0;
}
