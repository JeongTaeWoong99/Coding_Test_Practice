#include <bits/stdc++.h>
using namespace std;

int    n;               // 실수의 개수
double arr[10001];      // 입력받은 실수 배열
double ret, dp;         // ret : 전체 최댓값, dp : i번째에서 끝나는 연속 부분 최대 곱(= 누적값)

int main(int argc, char* argv[])
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    // 입력 처리
    cin >> n;
    for (int i = 0; i < n; i++) cin >> arr[i];

    // DP 초기화 (첫 번째 원소)
    dp  = arr[0];   // 시작 누적값
    ret = arr[0];   // 시작 최댓값

    // DP 로직 : i번째에서 끝나는 최대 곱 계산
    for (int i = 1; i < n; i++)
    {
        // 새로 시작 vs 이어서 곱하기 중 최댓값 선택
        if   (arr[i] > dp * arr[i]) dp  = arr[i];   // arr[i]로 새로 시작하는게, 누적값 * arr[i]보다 큼.
        else                        dp *= arr[i];   // 이어서 곱하기는게 더 큼.

        ret = max(dp, ret);  // 전체 최댓값 갱신
    }

    // 결과 출력 (소수점 셋째 자리까지)
    cout << fixed << setprecision(3) << ret << '\n';

    return 0;
}
