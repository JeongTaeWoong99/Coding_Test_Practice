#include <bits/stdc++.h>
using namespace std;

int n;
vector<string> v;

// 문자열 숫자 비교 함수 (큰 숫자 처리용)
bool CMP(string a, string b)
{
    if (a.size() != b.size())
        return a.size() < b.size();  // 길이가 다르면 짧은 게 작음

    return a < b;  // 길이가 같으면 사전순 비교
}

int main(int argc, char* argv[])
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    cin >> n;

    for (int i = 0; i < n; i++)
    {
        string temp;
        cin >> temp;
        temp += 'a';  // 마지막에 가상의 문자 추가 (숫자 강제 종료용)

        string numberS;

        for (char c : temp)
        {
            // 숫자인 경우
            if (c >= '0' && c <= '9')
            {
                numberS += c;  // 뒤에 더해주기
            }
            // 숫자가 아닌 경우 (마지막 가상 문자 포함)
            else
            {
                // 들어있던 값이 있으면
                if (!numberS.empty())
                {
                    // 앞의 0 제거
                    int idx = 0;
                    while (idx < numberS.size() - 1 && numberS[idx] == '0')
                        idx++;

                    v.push_back(numberS.substr(idx));  // 저장
                    numberS = "";  // 초기화
                }
            }
        }
    }

    sort(v.begin(), v.end(), CMP);  // 커스텀 비교 함수 사용

    for (string s : v)
        cout << s << '\n';

    return 0;
}
