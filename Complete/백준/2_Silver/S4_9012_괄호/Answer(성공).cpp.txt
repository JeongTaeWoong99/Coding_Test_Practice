#include <bits/stdc++.h>
using namespace std;

int    N;
string S;

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    cin >> N;

    for (int i = 0; i < N; i++)
    {
        stack<char> stk; // 매번 초기화

        cin >> S;

        for (char c : S)
        {
            // 비어있지 않고, 맨 위가 ')'일 때 => 무엇이든지 넣기
            if (!stk.empty() && stk.top() == ')')
            {
                stk.push(c);
            }
            // 비어있지 않고, 맨 위가 '('이면, ')'일 때 => 넣지 말고 pop
            else if (!stk.empty() && stk.top() == '(' && c == ')')
            {
                stk.pop();
            }
            // 넣기
            else
            {
                stk.push(c);
            }
        }

        if (stk.empty()) cout << "YES" << '\n';
        else             cout << "NO"  << '\n';
    }

    return 0;
}
