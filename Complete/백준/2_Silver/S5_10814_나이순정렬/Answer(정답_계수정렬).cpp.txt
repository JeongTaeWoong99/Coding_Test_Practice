#include <bits/stdc++.h>
using namespace std;

int    n, age;
string name;
vector<string> bucket[201];  // 나이별 버킷 (1~200)

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    cin >> n;

    // 입력받으면서 나이별 버킷에 저장 (가입 순서 자동 유지)
    for(int i = 0; i < n; i++)
    {
        cin >> age >> name;
        bucket[age].emplace_back(name);
    }

    // 나이 순서대로 출력 (1~200)
    for(int i = 1; i <= 200; i++)
    {
        for(const auto& s : bucket[i])
            cout << i << ' ' << s << '\n';
    }

    return 0;
}
