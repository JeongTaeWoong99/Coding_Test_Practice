#include <bits/stdc++.h>
using namespace std;

int    m, x;
int    S = 0;  // 집합을 비트로 표현 (1~20번 비트 사용)
string cmd;

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    cin >> m;

    while(m--)
    {
        cin >> cmd;

        // x를 집합에 추가 (x번 비트를 1로 설정)
        if(cmd == "add")
        {
            cin >> x;
            S |= (1 << x);
        }
        // x를 집합에서 제거 (x번 비트를 0으로 설정)
        else if(cmd == "remove")
        {
            cin >> x;
            S &= ~(1 << x);
        }
        // x가 있으면 1, 없으면 0 출력 (x번 비트 확인)
        else if(cmd == "check")
        {
            cin >> x;
            cout << ((S >> x) & 1) << "\n";
        }
        // x의 존재 여부 반전 (x번 비트 XOR)
        else if(cmd == "toggle")
        {
            cin >> x;
            S ^= (1 << x);
        }
        // 1~20 모두 추가 (모든 비트를 1로)
        else if(cmd == "all")
        {
            S = (1 << 21) - 1;
        }
        // "empty" 모두 제거 (모든 비트를 0으로)
        else
        {
            S = 0;
        }
    }

    return 0;
}
