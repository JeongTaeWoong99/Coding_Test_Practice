#include <bits/stdc++.h>
using namespace std;

int            n;
string         s;
vector<string> words;

// 비교 함수 : 길이 짧은 순 → 사전 순
bool cmp(string a, string b)
{
    // 길이가 다를 때, 길이로 비교
    if(a.length() != b.length())
        return a.length() < b.length();
    
    // 길이가 같을 때, 스트링 내부 char 값으로 비교
    return a < b;
}

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    // 입력 처리
    cin >> n;
    for(int i = 0; i < n; i++)
    {
        cin >> s;
        words.push_back(s);
    }

    // 정렬 : 길이 짧은 순 → 사전 순
    sort(words.begin(), words.end(), cmp);

    // 중복 제거하며 출력
    for(int i = 0; i < words.size(); i++)
    {
        if(i > 0 && words[i] == words[i-1]) continue;

        cout << words[i] << "\n";
    }

    return 0;
}
