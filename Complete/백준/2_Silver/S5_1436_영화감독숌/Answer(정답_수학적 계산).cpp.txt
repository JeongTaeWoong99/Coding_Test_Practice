#include <bits/stdc++.h>
using namespace std;

int n, temp, cnt;

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    cin >> n;

    // i = 666 시작
    for(int i = 666; ; i++)
    {
        temp = i;

        // 현재 숫자(i)에 "666"이 연속으로 포함되어 있는지 확인
        // 방법 : 수학적 계산으로 뒤에서 3자리씩 확인 (문자열 변환 없이)
        while(temp >= 666)
        {
            // temp의 뒤 3자리(% 1000)가 666인지 확인
            // 예 : 1666  % 1000 = 666 (○)
            // 예 : 66666 % 1000 = 666 (○)
            // 예 : 16660 % 1000 = 660 (×)
            if(temp % 1000 == 666)
            {
                cnt++;  // 종말의 수 발견
                break;  // 더 이상 확인할 필요 없음
            }

            // 앞자리로 이동 (맨 뒤 자리 제거)
            // 예 : 16660 → 1666 → 166 → 16 → 1
            temp /= 10;
        }

        // N번째 종말의 수를 찾았으면 출력 후 종료
        if(n == cnt)
        {
            cout << i << "\n";
            break;
        }
    }

    return 0;
}
