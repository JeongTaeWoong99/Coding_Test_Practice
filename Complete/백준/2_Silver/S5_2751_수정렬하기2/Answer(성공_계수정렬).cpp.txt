#include <bits/stdc++.h>
using namespace std;

// 계수 정렬(Counting Sort) 방식
// 메모리 : 약 2MB / 시간 : O(N + 범위)

const int OFFSET  = 1000000;  // 음수 → 양수 변환 보정값
const int MAX_VAL = 2000004;  // 배열 크기 (-1000000 ~ 1000000)

int  n, temp;
bool arr[MAX_VAL];

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    cin >> n;

    // 입력 받으면서 해당 인덱스를 true로 설정
    for (int i = 0; i < n; ++i)
    {
        cin >> temp;
        arr[temp + OFFSET] = true;  // 음수 보정 (예: -5 → arr[999995])
    }

    // 전체 범위를 순회하며 true인 값만 출력
    for (int i = 0; i < MAX_VAL; ++i)
    {
        if (arr[i])
            cout << (i - OFFSET) << '\n';  // 원래 값으로 복원
    }

    return 0;
}
