#include <bits/stdc++.h>
using namespace std;

// vector + sort() 방식
// 메모리 : 약 8MB / 시간 : O(N log N)

int         n, temp;
vector<int> v;

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    cin >> n;

    // 입력 받으면서 vector에 저장
    for (int i = 0; i < n; ++i)
    {
        cin >> temp;
        v.emplace_back(temp);
    }

    // STL sort() 함수로 정렬 (퀵소트/머지소트 기반)
    sort(v.begin(), v.end());

    // 정렬된 결과 출력
    for (auto val : v)
        cout << val << '\n';

    return 0;
}
