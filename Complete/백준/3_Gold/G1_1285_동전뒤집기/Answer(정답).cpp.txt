#include<bits/stdc++.h>
using namespace std;

const int INF = 987654321;

int    N;            // 동전판의 크기 (N x N)
int    arr[44];      // 각 행의 상태를 비트마스킹으로 저장 (T=1, H=0)
int    ret = INF;    // 최소 T(뒷면) 개수
string S;            // 입력받을 한 행의 문자열

// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// 🔍 비트마스킹 핵심 개념 이해하기
// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
//
// 📌 비트마스킹이란?
//    - 정수 하나로 여러 개의 0/1 상태를 표현하는 기법
//    - 각 비트(0 또는 1)가 하나의 상태를 나타냄
//
// 예시: "HHT" 문자열을 비트마스킹으로 표현
//
//    문자열  :  H    H    T     (왼쪽부터 읽음)
//    비트값  :  0    0    1     (H=0, T=1)
//    비트위치:  2^0  2^1  2^2   (1,   2,   4)
//
//    이진수 표현 : 00000100 (2진수)
//    십진수 표현 : 4 (10진수)
//
//    ✅ "HHT"를 정수 4로 저장!
//
// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// 🎯 비트 연산자 복습
// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
//
// | (OR)  : 비트 하나라도 1이면 1
//    예 : 0101 | 0011 = 0111 (5 | 3 = 7)
//
// & (AND) : 비트 둘 다 1이어야 1
//    예 : 0101 & 0011 = 0001 (5 & 3 = 1)
//
// ~ (NOT) : 비트 반전 (0→1, 1→0)
//    예 : ~0101 = 1010 (모든 비트 반전)
//
// << (Left Shift) : 비트를 왼쪽으로 이동 (2배씩 증가)
//    예 : 1 << 3 = 1000(2진수) = 8(10진수)
//
// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

// here : 현재 확인 중인 행 번호 (1번부터 N번까지)
// 각 행을 뒤집을지 말지 결정하는 재귀 함수 (2^N 가지 경우 탐색)
void GO(int here)
{
    // 모든 행에 대한 뒤집기 결정 완료
    if(here == N + 1)
    {
        int sum = 0;  // 현재 상태에서 T(뒷면)의 최소 개수

        // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
        // 🔍 각 열을 확인하는 루프
        // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
        //
        // i는 비트마스크로 각 열의 위치를 나타냄
        //
        // 예시 (N=3일 때):
        //    i = 1    (이진수: 001) → 1번째 열 확인 (2^0)
        //    i = 2    (이진수: 010) → 2번째 열 확인 (2^1)
        //    i = 4    (이진수: 100) → 3번째 열 확인 (2^2)
        //
        // (1 << (N-1)) = 2^(N-1)
        //    N=3이면 1<<2 = 4 → 마지막 열까지만 확인
        //
        // i *= 2는 다음 비트 위치로 이동 (1→2→4→8→...)
        // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

        for(int i = 1; i <= (1 << (N - 1)); i *= 2)
        {
            int cnt = 0;  // 현재 열에서 T(1)의 개수

            // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
            // 🔍 각 행을 순회하면서 현재 열의 비트가 1인지 확인
            // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
            //
            // arr[j] & i 연산의 의미:
            //
            // 예시 1: 1번 행 "HHT" (arr[1] = 4 = 0100)
            //         1번째 열(i=1) 확인
            //
            //    arr[1] : 0100  (4)
            //    i      : 0001  (1)
            //    ─────────────
            //    AND 결과: 0000  (0) → False (H)
            //
            // 예시 2: 1번 행 "HHT" (arr[1] = 4 = 0100)
            //         3번째 열(i=4) 확인
            //
            //    arr[1] : 0100  (4)
            //    i      : 0100  (4)
            //    ─────────────
            //    AND 결과: 0100  (4) → True (T)
            //
            // ✅ AND 연산 결과가 0이 아니면 해당 비트가 1(T)
            // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

            for(int j = 1; j <= N; j++)
            {
                if(arr[j] & i)  // j번째 행의 i번째 비트가 1(T)인지 확인
                    cnt++;
            }

            // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
            // 🎯 열을 뒤집을지 말지 그리디하게 결정
            // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
            //
            // cnt       : 현재 T의 개수 (열을 뒤집지 않음)
            // N - cnt   : 열을 뒤집었을 때 T의 개수
            //
            // 예시 (N=3):
            //    어떤 열이 [T, T, H]인 경우
            //    → cnt = 2 (T가 2개)
            //    → 뒤집으면 [H, H, T] → T가 1개
            //    → min(2, 3-2) = min(2, 1) = 1 선택
            //
            //    어떤 열이 [H, T, H]인 경우
            //    → cnt = 1 (T가 1개)
            //    → 뒤집으면 [T, H, T] → T가 2개
            //    → min(1, 3-1) = min(1, 2) = 1 선택
            //
            // ✅ 항상 더 적은 쪽을 선택하여 최소값 보장!
            // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

            sum += min(cnt, N - cnt);
        }

        ret = min(ret, sum);  // 최소값 갱신

        return;
    }

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // 🔄 재귀로 모든 경우의 수 탐색 (2^N)
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    //
    // 각 행마다 2가지 선택:
    //    1. here번째 행을 뒤집지 않음
    //    2. here번째 행을 뒤집음
    //
    // N개 행이 있으므로 총 2^N 가지 경우의 수
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    // 1. here번째 행을 뒤집지 않는 경우
    GO(here + 1);

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // 🔄 NOT 연산자(~)로 행 전체 뒤집기
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    //
    // ~ 연산자는 모든 비트를 반전시킴 (0→1, 1→0)
    //
    // 예시: "HHT" (arr[here] = 4 = 0100)를 뒤집으면
    //
    //    원본      : 00000100 (4)
    //    ~연산 후  : 11111011 (음수... 문제 없나?)
    //
    // ⚠️ 주의: ~연산은 모든 비트를 반전시키므로
    //          사용하지 않는 상위 비트도 1이 됨
    //
    // BUT 안전한 이유:
    //    - AND 연산 시 i는 N비트까지만 사용
    //    - 예: i=1,2,4 (N=3일 때)만 사용
    //    - 상위 비트는 AND 연산에 영향 없음!
    //
    // 예시로 확인:
    //    arr[1] = 4   (0100) → "HHT"
    //    ~arr[1]      (1011 하위 4비트만 보면) → "TTH"
    //
    //    i=1과 AND: 1011 & 0001 = 0001 (True, T)
    //    i=2와 AND: 1011 & 0010 = 0010 (True, T)
    //    i=4와 AND: 1011 & 0100 = 0000 (False, H)
    //
    //    ✅ 정확히 "TTH"로 뒤집힘!
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    // 2. here번째 행을 뒤집는 경우 (NOT 연산으로 모든 비트 반전)
    arr[here] = ~arr[here];
    GO(here + 1);
}

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(NULL); cout.tie(NULL);

    // 입력 처리
    cin >> N;

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // 🔄 문자열을 비트마스킹으로 변환하는 과정
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    //
    // 예시: "HHT" 문자열을 정수로 변환
    //
    // 초기 상태:
    //    arr[i] = 0 (이진수: 00000000)
    //    value  = 1 (이진수: 00000001)
    //
    // 1번째 문자 'H':
    //    j = 'H' → 아무 작업 안 함
    //    arr[i] = 0 (변화 없음)
    //    value *= 2 → value = 2
    //
    // 2번째 문자 'H':
    //    j = 'H' → 아무 작업 안 함
    //    arr[i] = 0 (변화 없음)
    //    value *= 2 → value = 4
    //
    // 3번째 문자 'T':
    //    j = 'T' → arr[i] |= value
    //    arr[i] = 0 | 4 = 4 (이진수: 00000100)
    //    value *= 2 → value = 8 (다음을 위해 준비)
    //
    // 최종 결과:
    //    arr[i] = 4 (이진수: 00000100)
    //
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // 🔍 OR 연산자(|=)의 의미
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    //
    // arr[i] |= value는 특정 비트를 1로 설정하는 연산
    //
    // 예시: value = 4일 때
    //
    //    기존 arr[i] : 00000000
    //    value      : 00000100 (4)
    //    ──────────────────────
    //    OR 결과     : 00000100 (4)
    //
    // 또 다른 예시: arr[i] = 5이고 value = 2일 때
    //
    //    기존 arr[i] : 00000101 (5)
    //    value       : 00000010 (2)
    //    ──────────────────────
    //    OR 결과     : 00000111 (7)
    //
    // ✅ OR 연산은 기존 비트를 유지하면서 새 비트만 추가!
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    for(int i = 1; i <= N; i++)
    {
        cin >> S;  // i번째 행의 동전 상태 (예 : "HHT")

        int value = 1;  // 비트 위치 (1, 2, 4, 8, ... 즉 2^0, 2^1, 2^2, ...)
        for (char j : S)
        {
            if(j == 'T')
                arr[i] |= value;  // T면 해당 비트를 1로 설정 (OR 연산)

           value *= 2;  // 다음 비트 위치로 이동 (왼쪽으로 shift)
                        // value *= 2는 value <<= 1과 동일
        }
    }

    // 알고리즘 실행 : 1번 행부터 시작
    GO(1);

    // 결과 출력 : 최소 T(뒷면) 개수
    cout << ret << "\n";

    return 0;
}
