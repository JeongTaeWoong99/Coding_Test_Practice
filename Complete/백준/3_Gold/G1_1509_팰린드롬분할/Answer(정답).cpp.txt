#include <bits/stdc++.h>
using namespace std;

const int INF = 987654321;

bool   isPalin[2504][2504]; // isPalin[i][len] : s[i..i+len-1]이 팰린드롬이면 true
int    dp[2504];            // dp[i] : s[i..n-1]을 팰린드롬으로 분할할 때 최솟값
string s;
int    n;

// here : 현재 시작 인덱스
// s[here..n-1]을 팰린드롬으로 분할하는 최솟값 반환
//
// 동작 원리 :
//   here에서 시작하는 팰린드롬 구간 s[here..here+len-1]을 하나 선택하고
//   → 나머지 s[here+len..n-1]의 최솟값(재귀) + 1 을 계산
//   → 가능한 모든 len 중 최솟값을 dp[here]에 저장
//
// 예시 (ABACABA, here=0) :
//   len=1 → "A"      + DP(1)=2  → 3
//   len=3 → "ABA"    + DP(3)=2  → 3
//   len=7 → "ABACABA"+ DP(7)=0  → 1  ← 최솟값!
int DP(int here)
{
    if(here == n)       return 0;           // 문자열 끝 도달 → 분할 0개 추가
    if(dp[here] != INF) return dp[here];    // 이미 계산한 위치 → 재사용 (메모이제이션)

    int &ret = dp[here];                    // dp[here]를 참조 → ret 수정 = dp[here] 수정

    for(int len = 1; here + len <= n; len++)
    {
        if(isPalin[here][len])
        {
            // s[here..here+len-1] 팰린드롬 1개 선택 (+1)
            // + 나머지 s[here+len..n-1]의 최솟값 재귀 계산
            ret = min(ret, DP(here + len) + 1);
        }
    }

    return ret;
}

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    cin >> s;
    
    n = s.size();

    // DP 전처리 : 팰린드롬 여부 계산 (짧은 길이부터)

    // 길이 1 : 항상 팰린드롬
    for(int i = 0; i < n; i++)
    {
        isPalin[i][1] = true;
    }

    // 길이 2 : 두 문자가 같으면 팰린드롬
    for(int i = 0; i < n - 1; i++)
    {
        if(s[i] == s[i + 1])
        {
            isPalin[i][2] = true;
        }
    }

    // 길이 3 이상 : 양 끝이 같고 안쪽 구간이 팰린드롬이면 팰린드롬
    for(int len = 3; len <= n; len++)
    {
        for(int i = 0; i + len <= n; i++)
        {
            if(s[i] == s[i + len - 1] && isPalin[i + 1][len - 2])
            {
                isPalin[i][len] = true;
            }
        }
    }

    // dp 전체를 INF로 초기화
    // INF = "아직 계산 안 됨" 표시 → dp[here] != INF 조건으로 메모이제이션 구분
    // dp[n]도 INF로 세팅되지만, DP()에서 here==n이면 0 반환하므로 문제 없음
    fill(dp, dp + n + 1, INF);

    // DP(0) : 인덱스 0부터 시작해서 전체 문자열 s[0..n-1]의 최솟값 반환
    // 내부에서 재귀 호출하며 dp[] 테이블을 채워나감
    // 한 번 계산된 dp[i]는 재사용되어 중복 계산 없음 (O(n²))
    cout << DP(0) << "\n";

    return 0;
}
