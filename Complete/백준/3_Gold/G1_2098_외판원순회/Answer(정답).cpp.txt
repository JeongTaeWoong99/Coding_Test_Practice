#include <bits/stdc++.h>
using namespace std;

#define MAX_N 16

const int INF = 987654321;

// 전역 변수
int n;                        // 도시의 개수
int dp[MAX_N][1 << MAX_N];    // dp[현재도시][방문상태] : 최소 비용
int dist[MAX_N][MAX_N];       // dist[i][j] : i번 도시에서 j번 도시로 가는 비용

// here : 현재 위치한 도시, visited : 방문한 도시들의 비트마스크
int TSP(int here, int visited)
{
    // 모든 도시를 방문했다면(== 연산자)
    if(visited == (1 << n) - 1)
    {
        // 현재 도시에서 시작 도시(0번)로 돌아가는 비용 반환
        return dist[here][0] ? dist[here][0] : INF;
    }

    // 참조로 가져오기
    int &ret = dp[here][visited];

    // 이미 계산된 값이 있으면 반환 (메모이제이션)
    if(ret != -1) return ret;

    // 최소 비용 계산
    ret = INF;
    for(int i = 0; i < n; i++)
    {
        if(visited & (1 << i)) continue;    // 이미 방문한 도시는 건너뛰기(& 연산자)

        if(dist[here][i] == 0) continue;    // 갈 수 없는 도시는 건너뛰기

        // i번 도시로 이동 후 나머지 도시들을 방문하는 최소 비용
        ret = min(ret, TSP(i, visited | (1 << i)) + dist[here][i]);
    }

    return ret;
}

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    // 입력 처리
    cin >> n;
    for(int i = 0; i < n; i++)
    {
        for(int j = 0; j < n; j++)
        {
            cin >> dist[i][j];
        }
    }

    // DP 테이블 초기화
    memset(dp, -1, sizeof(dp));

    // 0번 도시에서 출발 (0번 도시를 방문한 상태로 시작)
    cout << TSP(0, 1);

    return 0;
}
