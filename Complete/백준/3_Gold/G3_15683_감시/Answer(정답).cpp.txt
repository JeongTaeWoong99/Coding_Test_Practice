#include <bits/stdc++.h>
using namespace std;

// 상, 우, 하, 좌 방향 벡터 (시계방향)
const int dy[] = {-1, 0, 1, 0};
const int dx[] = {0, 1, 0, -1};

int                      n, m;            // 사무실 세로/가로 크기
int                      a[10][10];       // 사무실 맵 (0:빈칸, 1~5:CCTV, 6:벽, 8:감시영역)
int                      ret = 987654321; // 사각지대 최솟값
vector<pair<int, int>>   v;               // CCTV 위치 저장

// here : CCTV 인덱스
// dir : 방향 (0:상, 1:우, 2:하, 3:좌)
// CCTV가 감시하는 영역을 표시하고 변경된 좌표 반환
vector<pair<int, int>> ExtendCCTV(int here, int dir)
{
    vector<pair<int, int>> _change;
    int y = v[here].first;
    int x = v[here].second;

    // 1번 CCTV : 한 방향 감시
    if(a[y][x] == 1)
    {
        // 1방향만 하면 되니 for문 불필요
        while(true)
        {
            int ny = y + dy[dir];
            int nx = x + dx[dir];

            if(ny >= 0 && ny < n && nx >= 0 && nx < m && a[ny][nx] != 6)
            {
                if(a[ny][nx] == 0)
                {
                    a[ny][nx] = 8;                      // 감시 영역 표시
                    _change.emplace_back(ny, nx); // 변경 좌표 기록
                }
                y = ny;
                x = nx;
            }
            else
                break;
        }
    }
    // 2번 CCTV : 양 방향 감시 (180도 반대)
    else if(a[y][x] == 2)
    {
        // i += 2를 통해, 180도 반대 확인
        for(int i = 0; i <= 2; i += 2)
        {
            // 오리지널 값 회손 방지용
            int _y = y;
            int _x = x;
            
            while(true)
            {
                int ny = _y + dy[(dir + i) % 4];
                int nx = _x + dx[(dir + i) % 4];

                if(ny >= 0 && ny < n && nx >= 0 && nx < m && a[ny][nx] != 6)
                {
                    if(a[ny][nx] == 0)
                    {
                        a[ny][nx] = 8;                // 감시 영역 표시
                        _change.emplace_back(ny, nx); // 변경 좌표 기록
                    }
                    _y = ny;
                    _x = nx;
                }
                else
                    break;
            }
        }
    }
    // 3번 CCTV : 직각 방향 감시 (90도)
    else if(a[y][x] == 3)
    {
        // 직각이니, i++
        for(int i = 0; i < 2; i++)
        {
            // 오리지널 값 회손 방지용
            int _y = y;
            int _x = x;
            while(true)
            {
                int ny = _y + dy[(dir + i) % 4];
                int nx = _x + dx[(dir + i) % 4];

                if(ny >= 0 && ny < n && nx >= 0 && nx < m && a[ny][nx] != 6)
                {
                    if(a[ny][nx] == 0)
                    {
                        a[ny][nx] = 8;                // 감시 영역 표시
                        _change.emplace_back(ny, nx); // 변경 좌표 기록
                    }
                    _y = ny;
                    _x = nx;
                }
                else
                    break;
            }
        }
    }
    // 4번 CCTV : 세 방향 감시
    else if(a[y][x] == 4)
    {
        // 3방향이니 i++ 
        for(int i = 0; i < 3; i++)
        {
            // 오리지널 값 회손 방지용
            int _y = y;
            int _x = x;
            while(true)
            {
                int ny = _y + dy[(dir + i) % 4];
                int nx = _x + dx[(dir + i) % 4];

                if(ny >= 0 && ny < n && nx >= 0 && nx < m && a[ny][nx] != 6)
                {
                    if(a[ny][nx] == 0)
                    {
                        a[ny][nx] = 8;                      // 감시 영역 표시
                        _change.emplace_back(ny, nx); // 변경 좌표 기록
                    }
                    _y = ny;
                    _x = nx;
                }
                else
                    break;
            }
        }
    }
    // 5번 CCTV : 네 방향 감시
    else if(a[y][x] == 5)
    {
        // 전체 방향 확인 i++
        for(int i = 0; i < 4; i++)
        {
            // 오리지널 값 회손 방지용
            int _y = y;
            int _x = x;
            while(true)
            {
                int ny = _y + dy[(dir + i) % 4];
                int nx = _x + dx[(dir + i) % 4];

                if(ny >= 0 && ny < n && nx >= 0 && nx < m && a[ny][nx] != 6)
                {
                    if(a[ny][nx] == 0)
                    {
                        a[ny][nx] = 8;                // 감시 영역 표시
                        _change.emplace_back(ny, nx); // 변경 좌표 기록
                    }
                    _y = ny;
                    _x = nx;
                }
                else
                    break;
            }
        }
    }

    return _change;
}

// here : 현재 처리 중인 CCTV 인덱스
// 백트래킹으로 모든 CCTV 방향 조합 탐색
void DFS(int here)
{
    // 모든 CCTV 방향 설정 완료 시 사각지대 계산
    // 기저 사례
    if(here == v.size())
    {
        int cnt = 0;
        for(int i = 0; i < n; i++)
        {
            for(int j = 0; j < m; j++)
            {
                // 빈 칸 개수 세기
                if(a[i][j] == 0) 
                    cnt++;
            }
        }
        
        ret = min(cnt, ret); // 최솟값 갱신
        
        return;
    }

    // 현재 CCTV의 4가지 방향 시도
    // DFS 재귀 부분
    for(int k = 0; k < 4; k++)
    {
        vector<pair<int, int>> _change = ExtendCCTV(here, k);  // 감시 영역 표시
        
        DFS(here + 1);                                         // 다음 CCTV 처리
        
        for(auto b : _change)                                  // 백트래킹 (원상복구)
            a[b.first][b.second] = 0;
    }
}

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    // 입력 처리
    cin >> n >> m;
    for(int i = 0; i < n; i++)
    {
        for(int j = 0; j < m; j++)
        {
            cin >> a[i][j];

            // CCTV 위치 저장
            if(a[i][j] != 6 && a[i][j] != 0)
                v.emplace_back(i, j);
        }
    }

    // 백트래킹으로 모든 경우의 수 탐색(CCTV 위치 0부터 시작)
    DFS(0);

    // 결과 출력
    cout << ret << "\n";

    return 0;
}
