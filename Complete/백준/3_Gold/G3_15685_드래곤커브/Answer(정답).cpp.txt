#include<bits/stdc++.h>
using namespace std;

// 방향 배열 : 0(→), 1(↑), 2(←), 3(↓)
const int dy[] = {0, -1, 0, 1};  // y 좌표 변화량
const int dx[] = {1, 0, -1, 0};  // x 좌표 변화량

vector<int> dragon[4][11];       // dragon[시작방향][세대] : 해당 드래곤 커브의 방향 시퀀스
int n;                           // 드래곤 커브의 개수
int x, y, d, g;                  // x, y : 시작점, d : 시작방향, g : 세대
int a[101][101];                 // 격자판 : 드래곤 커브가 지나간 점은 1
int cnt;                         // 1x1 정사각형 개수

// x : 시작 x좌표, y : 시작 y좌표, d : 시작 방향, g : 세대
// 주어진 시작점과 방향으로 g세대 드래곤 커브를 그림
void GO(int x, int y, int d, int g)
{
    a[x][y] = 1;                          // 시작점 방문 표시

    // 0세대부터 g세대까지 순차적으로 그림
    for(int i = 0; i <= g; i++)
    {
        // 현재 세대의 방향 시퀀스를 따라 이동
        for(int dir : dragon[d][i])
        {
            x += dx[dir];                 // x 좌표 이동
            y += dy[dir];                 // y 좌표 이동
            a[x][y] = 1;                  // 이동한 점 방문 표시
        }
    }
    return;
}

// 모든 시작방향(0~3)과 세대(0~10)에 대한 드래곤 커브의 방향 시퀀스를 미리 생성
// 핵심 규칙 : K세대 = (K-1세대를 역순으로 +1(시계방향 90도 회전)) + K-1세대
void MakeDragon()
{
    // 4가지 시작 방향에 대해
    for(int i = 0; i < 4; i++)
    {
        dragon[i][0].push_back(i);                          // 0세대 : 시작 방향 하나
        dragon[i][1].push_back((i + 1) % 4);                // 1세대 추가 부분 : 시작방향 + 90도 회전

        // 2세대부터 10세대까지 생성
        for(int j = 2; j <= 10; j++)
        {
            int n = dragon[i][j - 1].size();                // 이전 세대 크기

            // [새로 추가되는 부분] : 이전 세대를 역순으로 순회하며 +1 (시계방향 90도)
            for(int k = n - 1; k >= 0; k--)
            {
                dragon[i][j].push_back((dragon[i][j - 1][k] + 1) % 4);
            }

            // [기존 부분] : 이전 세대 그대로 복사
            for(int k = 0; k < n; k++)
            {
                dragon[i][j].push_back(dragon[i][j - 1][k]);
            }
        }
    }
    return;
}

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    // 입력 처리
    cin >> n;
    MakeDragon();                    // 모든 드래곤 커브 패턴 미리 생성

    // n개의 드래곤 커브 그리기
    for(int i = 0; i < n; i++)
    {
        cin >> x >> y >> d >> g;
        GO(x, y, d, g);              // 격자판에 드래곤 커브 표시
    }

    // 1x1 정사각형 개수 세기
    // 정사각형의 왼쪽 위 꼭짓점 (i, j) 기준으로 체크
    for(int i = 0; i <= 100; i++)
    {
        for(int j = 0; j <= 100; j++)
        {
            // 네 꼭짓점이 모두 드래곤 커브의 일부인지 확인
            if(a[i][j] && a[i + 1][j] && a[i + 1][j + 1] && a[i][j + 1])
                cnt++;
        }
    }

    // 결과 출력
    cout << cnt << "\n";

    return 0;
}
