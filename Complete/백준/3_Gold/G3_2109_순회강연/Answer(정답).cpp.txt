#include <bits/stdc++.h>
using namespace std;

// 전역 변수
int                                            n;      // 대학 개수
int                                            a, b;   // a : 강연료, b : 마감일
int                                            ret;    // 최대 수익
vector<pair<int, int>>                         v;      // {마감일, 강연료} 쌍 저장
priority_queue<int, vector<int>, greater<int>> minPQ;  // 최소 힙 (가장 적은 강연료가 top)

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    // 입력 처리
    cin >> n;
    for(int i = 0; i < n; i++)
    {
        cin >> a >> b;                 // a : 강연료, b : 마감일
        v.emplace_back(b, a);    // {마감일, 강연료} 형태로 저장
    }

    // 마감일 기준 오름차순 정렬
    sort(v.begin(), v.end());

    // 그리디 알고리즘 : 마감일이 빠른 것부터 처리
    for(int i = 0; i < n; i++)
    {
        minPQ.push(v[i].second);    // 현재 강연료를 우선순위 큐에 추가

        // 선택한 강연 수가 현재 마감일을 초과하면
        // 가장 적은 강연료를 제거 (더 좋은 강연을 선택하기 위해)
        if(minPQ.size() > v[i].first)
        {
            minPQ.pop();            // 최소값 제거
        }
    }

    // 최종 선택된 강연료들의 합 계산
    while(!minPQ.empty())
    {
        ret += minPQ.top();
        minPQ.pop();
    }

    cout << ret << "\n";

    return 0;
}
