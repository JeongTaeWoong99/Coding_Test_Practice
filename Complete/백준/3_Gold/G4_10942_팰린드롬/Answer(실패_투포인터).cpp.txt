#include <bits/stdc++.h>
using namespace std;

int n, m;
int leftN, rightN;
int arr[2004];  // 수열 (1-indexed)

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    // 수열 입력
    cin >> n;
    for (int i = 1; i <= n; ++i)
        cin >> arr[i];

    // m번 쿼리 : 매 쿼리마다 양쪽에서 포인터를 이동하며 팰린드롬 확인
    cin >> m;
    while (m--)
    {
        cin >> leftN >> rightN;

        int moveNum = 0; // 좌우 이동 횟수

        while (true)
        {
            // 양 끝 원소가 다르면 팰린드롬 X
            if (arr[leftN + moveNum] != arr[rightN - moveNum])
            {
                cout << "0\n";
                break;
            }

            // 포인터가 만나거나 교차 → 모두 같음 → 팰린드롬 O
            if (leftN + moveNum >= rightN - moveNum)
            {
                cout << "1\n";
                break;
            }

            // 안쪽으로 한 칸씩 이동
            moveNum++;
        }
    }

    return 0;
}

// [실패 원인]
// 이 방법의 시간복잡도는 O(M x N/2) = 최악 1,000,000 x 1,000 = 10억
// 시간 제한 0.5초를 훨씬 초과 → TLE (시간 초과)
//
// 투포인터 자체의 로직은 올바르지만, M이 최대 100만이므로
// 매 쿼리마다 O(N) 탐색을 반복하면 감당할 수 없음.
// → O(N^2) 전처리 + O(1) 쿼리 응답인 DP 방식이 필요
