#include <bits/stdc++.h>
using namespace std;

int            n, num, len;
int            lis[1004];
pair<int, int> ans[1004];  // {LIS에서의 위치, 실제 값} 저장
stack <int>    stk;

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    cin >> n;

    // n <= 1000 보다 큰 값으로 초기화
    fill_n(lis, 1004, 2000);

    for (int i = 0; i < n; i++)
    {
        cin >> num;

        // 위치 찾기 (주소 - 주소 = 인덱스)
        int pos = lower_bound(lis, lis + len, num) - lis;

        // 새 위치 = 길이 증가
        if (lis[pos] == 2000) len++;

        // 값 넣어주기
        lis[pos] = num;

        // 역추적을 위해 저장 : {LIS에서의 위치, 실제 값}
        ans[i].first  = pos;
        ans[i].second = num;
    }

    // LIS 길이 출력
    cout << len << "\n";

    // 역추적 : 뒤에서부터 LIS 복원
    for(int i = n - 1; i >= 0; i--)
    {
        // len - 1 = 길이 보정
        if(ans[i].first == len - 1)
        {
            stk.push(ans[i].second);
            len--;
        }
    }

    // 스택에서 꺼내며 출력 (오름차순)
    while(!stk.empty())
    {
        cout << stk.top() << " ";
        stk.pop();
    }

    return 0;
}
