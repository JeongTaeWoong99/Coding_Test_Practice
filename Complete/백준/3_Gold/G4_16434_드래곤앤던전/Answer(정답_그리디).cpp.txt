#include<bits/stdc++.h>
using namespace std;

typedef long long ll;

ll n, atk;                   // n : 방의 개수, atk : 초기 공격력
ll t, a, h;                  // t : 방 타입 (1:몬스터, 2:포션), a : 공격력/증가량, h : 체력/회복량
ll currentHP, needHP;        // currentHP : 현재 체력, needHP : 필요한 최대 체력
ll damage;                   // 몬스터와 싸울 때 받는 데미지

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);

    cin >> n >> atk;

    for (int i = 0; i < n; i++)
    {
        cin >> t >> a >> h;

        // 몬스터 방
        if (t == 1)
        {
            // 받을 데미지 계산
            // ceil((double)h/atk) : 몬스터를 죽이는데 필요한 공격 횟수를 반올림 = 3/2 = 1.5 = 2
            // 0.xxx 소수가 있다는건 1번 더 공격해야 한다는 것 이기 때문에, 반올림 하는 것과 동일
            // (공격 횟수 - 1) : 내가 받는 공격 횟수
            // a * (공격 횟수 - 1) : 총 받는 데미지
            damage = a * (ceil((double)h / atk) - 1);

            // 현재 체력이 데미지보다 적으면
            if (currentHP < damage)
            {
                // 부족한 체력만큼 최대 체력 증가
                needHP += damage - currentHP;
                currentHP = 0;
            }
            // 현재 체력이 충분하면
            else
            {
                // 데미지만큼 현재 체력 감소
                currentHP -= damage;
            }
        }
        // 포션 방
        else
        {
            // 공격력 증가
            atk += a;
            // 체력 회복 (최대 체력 초과 불가)
            currentHP = min(currentHP + h, needHP);
        }
    }

    // 최대 체력 + 1 출력 (체력은 1부터 시작)
    cout << needHP + 1 << "\n";

    return 0;
}
