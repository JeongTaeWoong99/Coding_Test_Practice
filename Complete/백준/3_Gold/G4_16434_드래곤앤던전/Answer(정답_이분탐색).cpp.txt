#include<bits/stdc++.h>
using namespace std;

typedef long long ll;

ll n, attack, t[130005], a[130005], h[130005], lo, hi, ret;

bool check(ll mid)
{
    ll mxHP        = mid;         // 최대 체력
    ll init_attack = attack;      // 현재 공격력

    for (int i = 0; i < n; i++)
    {
        // 포션 방
        if (t[i] == 2)
        {
            // 체력 회복 (최대 체력 초과 불가)
            mid = min(mxHP, mid + h[i]);
            // 공격력 증가
            init_attack += a[i];
        }
        // 몬스터 방
        else
        {
            // 몬스터 처치에 필요한 공격 횟수 계산
            ll cnt = h[i] / init_attack + (h[i] % init_attack ? 1 : 0);
            // 내 공격 횟수 - 1 만큼 적의 공격을 받음
            mid -= (cnt - 1) * a[i];
        }

        // 중간에 죽으면 즉시 false 반환
        if (mid <= 0) return false;
    }

    return true;
}

int main()
{
    ios::sync_with_stdio(false);cin.tie(NULL);cout.tie(NULL);

    cin >> n >> attack;

    for (int i = 0; i < n; i++)
    {
        cin >> t[i] >> a[i] >> h[i];
    }

    lo = 1;
    hi = 1e18 + 4;

    while (lo <= hi)
    {
        ll mid = (lo + hi) / 2;

        if (check(mid))
        {
            hi  = mid - 1;
            ret = mid;
        }
        else
        {
            lo = mid + 1;
        }
    }

    cout << ret << '\n';

    return 0;
}
