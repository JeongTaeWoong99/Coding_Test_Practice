#include <bits/stdc++.h>
using namespace std;

int         N;
int         arr[1000004], ret[1000004];
stack<int>  stk;                        // 인덱스를 저장하는 스택

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(NULL);cout.tie(NULL);

    cin >> N;
    memset(ret, -1, sizeof(ret));

    for (int i = 0; i < N; i++)
    {
        cin >> arr[i];
        
        // ★ top pop while 반복!
        // ★ arr[stk.top()] < arr[i] // stk.pop()
        // 스택이 비어있지 않고, 스택의 top 인덱스의 값보다 현재 값이 크면
        while (!stk.empty() && arr[stk.top()] < arr[i])
        {
            ret[stk.top()] = arr[i];  // 오큰수 설정
            stk.pop();                // 스택에서 제거
        }

        stk.push(i);  // 현재 인덱스를 스택에 추가
    }

    // 결과 출력
    for (int i = 0; i < N; i++)
        cout << ret[i] << " ";

    return 0;
}
