#include <bits/stdc++.h>
using namespace std;

const int INF = 1e9;

int n, ans = INF;
int cost[1001][3]; // cost[i][c] : i번 집을 c색으로 칠하는 비용 (0:R, 1:G, 2:B)
int dp[1001][3];   // dp[i][c]   : i번 집을 c색으로 칠할 때의 최솟값

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    // 입력
    cin >> n;
    for (int i = 0; i < n; ++i)
    {
        cin >> cost[i][0] >> cost[i][1] >> cost[i][2];
    }

    // 1번 집의 색을 고정하고 DP 3번 수행
    // 원형 조건 : 1번 집과 N번 집의 색이 달라야 함
    for (int first = 0; first < 3; ++first)
    {
        // 1번 집 초기화 : 고정된 색만 유효, 나머지는 INF
        for (int i = 0; i < 3; ++i)
        {
            if (i == first)
            {
                dp[0][i] = cost[0][i];
            }
            else
            {
                dp[0][i] = INF;
            }
        }

        // 2번 ~ N번 집 DP 전이
        for (int i = 1; i < n; ++i)
        {
            for (int j = 0; j < 3; ++j)
            {
                dp[i][j] = INF;

                // 이전 집과 다른 색 중 최솟값 선택
                for (int prev = 0; prev < 3; ++prev)
                {
                    if (prev != j)
                    {
                        dp[i][j] = min(dp[i][j], dp[i-1][prev] + cost[i][j]);
                    }
                }
            }
        }

        // N번 집은 1번 집과 다른 색이어야 함
        for (int last = 0; last < 3; ++last)
        {
            if (last != first)
            {
                ans = min(ans, dp[n-1][last]);
            }
        }
    }

    // 결과 출력
    cout << ans << "\n";

    return 0;
}
