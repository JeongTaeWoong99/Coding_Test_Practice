#include<bits/stdc++.h>
using namespace std;

int n, m;              // n : 자두가 떨어지는 시간(초), m : 최대 이동 횟수
int drop[1004];        // drop[i] : i초에 자두가 떨어지는 나무 번호 (1 또는 2 => 0 또는 1로 변환)
int dp[1004][2][34];   // [현재 시간][현재 나무 위치(0 or 1)][남은 이동 횟수]

// TopDown DP (재귀 + 메모이제이션)
// curTime : 현재 시간, curTree : 현재 나무(0 or 1), remainMove : 남은 이동 횟수
int Go(int curTime, int curTree, int remainMove)
{
    // 남은 이동 횟수가 음수면 불가능한 경우 → 매우 작은 값 반환
    // (max 연산에서 선택되지 않도록)
    if(remainMove < 0) return -1e9;

    // 종료 조건 : 모든 시간이 지나면 0 반환
    // 이 문제는 m번 "이하"로 이동 가능 (정확히 m번 사용할 필요 없음)
    // ※ 만약 정확히 m번 사용해야 한다면 : if(curTime == n) return remainMove == 0 ? 0 : -1e9;
    if(curTime == n) return 0;

    // 메모이제이션 : 이미 계산된 값이 있으면 반환
    int &ret = dp[curTime][curTree][remainMove];
    if(ret != -1) return ret;

    // 점화식 : 안 움직이기 vs 움직이기 중 최댓값
    // curTree^1 : XOR 연산으로 0 ↔ 1 토글 (0 → 1, 1 → 0)
    // + (curTree == drop[curTime]) 의미 : 현재 나무에 자두가 떨어지면, == 으로 비교하여 같으면, true는 1을 반환
    return ret = max(Go(curTime + 1, curTree  , remainMove    ),  // 안 움직임
                     Go(curTime + 1, curTree^1, remainMove - 1))  // 다른 나무로 이동
                     + (curTree == drop[curTime]); // 두 움직임 중, 더 큰값을 사용하고, 현재 자리도 비교하여, +1 해줌
}

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    // 입력 : n초 동안, m번 이동 가능
    cin >> n >> m;

    // 각 초마다 떨어지는 나무 번호 입력 (1 또는 2)
    // ★ 비교 시 -1 해서 0 또는 1로 변환하여 사용 => 토글 연산자 사용을 위해서!
    for(int i = 0; i < n; i++)
    {
        cin >> drop[i];
        drop[i]--;      // 1 → 0, 2 → 1 변환
    }

    // DP 배열 초기화 (-1 : 아직 계산 안 됨)
    memset(dp, -1, sizeof(dp));

    // 탑다운 DP 시작
    // 자두는 처음에 1번 나무(0) 아래에 있음
    // 경우 1 : 1번 나무(0)에 그대로 있기 → 이동 0번
    // 경우 2 : 2번 나무(1)로 이동하기    → 이동 1번
    cout << max(Go(0, 0, m), Go(0, 1, m - 1));

    return 0;
}
