#include<bits/stdc++.h>
using namespace std;

int n, m;              // n : 자두가 떨어지는 시간(초), m : 최대 이동 횟수
int drop[1004];        // drop[i] : i초에 자두가 떨어지는 나무 번호 (1 또는 2)
int dp[1004][3][34];   // [현재 시간][현재 나무 위치(1 or 2)][남은 이동 횟수]

// TopDown DP (재귀 + 메모이제이션)
// curTime : 현재 시간, curTree : 현재 나무(1 or 2), remainMove : 남은 이동 횟수
int Go(int curTime, int curTree, int remainMove)
{
    // 남은 이동 횟수가 음수면 불가능한 경우 → 매우 작은 값 반환
    // (max 연산에서 선택되지 않도록)
    if(remainMove < 0) return -1e9;

    // 종료 조건 : 모든 시간이 지나면 0 반환
    if(curTime == n) return 0;

    // 메모이제이션 : 이미 계산된 값이 있으면 반환
    int &ret = dp[curTime][curTree][remainMove];
    if(ret != -1) return ret;

    // ❌ 토글 연산자 미사용 버전 ❌
    // curTree^1 대신 if-else로 다른 나무 위치 계산
    int otherTree;
    if(curTree == 1)
        otherTree = 2;
    else
        otherTree = 1;

    // 점화식 : 안 움직이기 vs 움직이기 중 최댓값
    // (curTree == drop[curTime]) : 현재 나무에 자두가 떨어지면 +1
    return ret = max(Go(curTime + 1, curTree   , remainMove    ),                            // 안 움직임
                     Go(curTime + 1, otherTree , remainMove - 1)) + (curTree == drop[curTime]); // 다른 나무로 이동
}

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    // 입력 : n초 동안, m번 이동 가능
    cin >> n >> m;

    // 각 초마다 떨어지는 나무 번호 입력 (1 또는 2)
    for(int i = 0; i < n; i++)
        cin >> drop[i];

    // DP 배열 초기화 (-1 : 아직 계산 안 됨)
    memset(dp, -1, sizeof(dp));

    // 탑다운 DP 시작
    // 자두는 처음에 1번 나무 아래에 있음
    // 경우 1 : 1번 나무에 그대로 있기 → 이동 0번
    // 경우 2 : 2번 나무로 이동하기 → 이동 1번
    cout << max(Go(0, 1, m), Go(0, 2, m - 1));

    return 0;
}
