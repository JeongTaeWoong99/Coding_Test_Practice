#include <bits/stdc++.h>
using namespace std;

const int dY[4] = {-1, 0, 1, 0};
const int dX[4] = {0, 1, 0, -1};

int R, C;
int arr[1004][1004];
int visitedFire[1004][1004];
int visitedJihoon[1004][1004];
queue<pair<int, int>> fireQ;
queue<pair<int, int>> jihoonQ;

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    cin >> R >> C;

    // 입력 받기
    for (int i = 0; i < R; i++)
    {
        string temp;
        cin >> temp;

        for (int j = 0; j < C; j++)
        {
            arr[i][j] = temp[j];

            if (arr[i][j] == 'J')
            {
                jihoonQ.emplace(i, j);
                visitedJihoon[i][j] = 1;  // 초기 위치 방문 처리
            }
            else if (arr[i][j] == 'F')
            {
                fireQ.emplace(i, j);
                visitedFire[i][j] = 1;    // 초기 위치 방문 처리
            }
        }
    }

    // 1단계: 불 확산 시간 기록 (멀티소스 BFS)
    while (!fireQ.empty())
    {
        int fireY = fireQ.front().first;
        int fireX = fireQ.front().second;
        fireQ.pop();

        for (int i = 0; i < 4; i++)
        {
            int nY = fireY + dY[i];
            int nX = fireX + dX[i];

            // 범위 체크
            if (nY < 0 || nY >= R || nX < 0 || nX >= C) continue;

            // 벽 체크
            if (arr[nY][nX] == '#') continue;

            // 처음 방문
            if (visitedFire[nY][nX] == 0)
            {
                visitedFire[nY][nX] = visitedFire[fireY][fireX] + 1;  // 시간 누적
                fireQ.emplace(nY, nX);
            }
        }
    }

    // 2단계: 지훈이 탈출 경로 탐색 (단일소스 BFS)
    while (!jihoonQ.empty())
    {
        int jihoonY = jihoonQ.front().first;
        int jihoonX = jihoonQ.front().second;
        jihoonQ.pop();

        // 현재 위치가 가장자리면 탈출 성공!
        if (jihoonY == 0 || jihoonY == R - 1 || jihoonX == 0 || jihoonX == C - 1)
        {
            cout << visitedJihoon[jihoonY][jihoonX];
            return 0;
        }

        for (int i = 0; i < 4; i++)
        {
            int nY = jihoonY + dY[i];
            int nX = jihoonX + dX[i];

            // 범위 체크
            if (nY < 0 || nY >= R || nX < 0 || nX >= C) continue;

            // 벽 체크
            if (arr[nY][nX] == '#') continue;

            // 지훈 처음 방문 && (불 도달 안함 || 지훈이 불보다 빨리 도착)
            if (visitedJihoon[nY][nX] == 0 &&
                (visitedFire[nY][nX] == 0 || visitedJihoon[jihoonY][jihoonX] + 1 < visitedFire[nY][nX]))
            {
                visitedJihoon[nY][nX] = visitedJihoon[jihoonY][jihoonX] + 1;  // 시간 누적
                jihoonQ.emplace(nY, nX);
            }
        }
    }

    // 탈출 불가능
    cout << "IMPOSSIBLE";

    return 0;
}
