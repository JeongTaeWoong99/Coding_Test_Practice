using namespace std;

int  n, m;       // n : 목표 채널, m : 고장난 버튼 수
bool broken[10]; // broken[d] = true이면 숫자 d 버튼이 고장

// ch의 각 자리 숫자가 모두 고장나지 않은 버튼인지 확인
// ex) ch=545, broken={6,7,8} → '5','4','5' 모두 정상 → true
// ex) ch=567, broken={6,7,8} → '6' 고장 → false
bool CanType(int ch)
{
    string s = to_string(ch); // 숫자를 문자열로 변환 (각 자리 순회 위해)

    for (char c : s)
    {
        if (broken[c - '0']) // 해당 자리 숫자가 고장났으면 입력 불가
        {
            return false;
        }
    }

    return true;
}

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    cin >> n >> m;

    for (int i = 0; i < m; i++)
    {
        int x;
        cin >> x;

        broken[x] = true;
    }

    // 경우 1 : +/-만 사용하는 경우 (현재 채널 100에서 이동 => 절대값)
    int ans = abs(n - 100);

    // 경우 2 : 숫자 버튼으로 채널 직접 입력 후, +/-로 이동하는 경우
    // ch를 0~1000000까지 완전탐색하며 입력 가능한 채널을 모두 후보로 검사
    for (int ch = 0; ch <= 1000000; ch++)
    {
        if (CanType(ch)) // ch를 고장나지 않은 버튼으로 입력 가능한 경우
        {
            // 채널 입력에 필요한 버튼 수
            int digits = (int)to_string(ch).length();

            // 현재 선택된 ans 값과 비교하여, 더 낮으면 교체
            // digits(숫자 입력) + abs(n - ch)(+/-로 나머지 이동) 중 최솟값 유지
            ans = min(ans, digits + abs(n - ch));
        }
    }

    cout << ans << "\n";

    return 0;
}
