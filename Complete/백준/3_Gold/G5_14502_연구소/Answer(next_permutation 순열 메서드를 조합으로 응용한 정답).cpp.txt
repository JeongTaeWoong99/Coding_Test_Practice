// 백준 14502번 - 연구소 (next_permutation 순열 메서드를 조합으로 응용)
// N X M (3 ≤ N, M ≤ 8)
// 빈칸 = 0, 벽 = 1, 바이러스 = 2
// 상하좌우 인접 이동
// 새로운 벽 3개 무조건 세워야 함
//
// ⭐ 핵심: next_permutation은 "순열" 메서드지만, 0/1 선택 마스크를 사용하면 "조합"으로 응용 가능!
// - {0,0,0,1,1}의 순열 = 5개 중 3개를 선택하는 모든 조합
// - 0의 위치 = 선택된 항목
// - 1의 위치 = 선택되지 않은 항목
// - 같은 값(0끼리, 1끼리)의 순서는 구분되지 않으므로 중복 없는 조합 생성
// - 생성되는 개수: 5!/(3!×2!) = C(5,3) = 10가지

#include <bits/stdc++.h>
using namespace std;

const int dy[] = {-1, 0, 1,  0};
const int dx[] = { 0, 1, 0, -1};

int arr[10][10];
int visited[10][10];
int N, M, ret;
vector<pair<int, int>> virusList;
vector<pair<int, int>> emptyList;

void DFS(int y, int x)
{
    for(int i = 0; i < 4; i++)
    {
        int ny = y + dy[i];
        int nx = x + dx[i];

        // 경계 / 방문 / 벽 체크
        if(ny < 0 || ny >= N || nx < 0 || nx >= M || visited[ny][nx] || arr[ny][nx] == 1) continue;

        visited[ny][nx] = 1; // 감염 표시(방문)

        DFS(ny, nx);
    }
}

int Check()
{
    // 방문 배열 초기화
    fill_n(&visited[0][0], 10 * 10, 0);

    // 모든 바이러스에서 확산 시작
    for(pair<int, int> virus : virusList)
    {
        visited[virus.first][virus.second] = 1;
        DFS(virus.first, virus.second);
    }

    // 안전영역(0이면서 방문되지 않은 칸) 카운트
    int cnt = 0;
    for(int i = 0; i < N; i++)
    {
        for(int j = 0; j < M; j++)
        {
            if(arr[i][j] == 0 && !visited[i][j]) cnt++;
        }
    }
    return cnt;
}

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    cin >> N >> M;

    // 입력 파싱 및 빈칸/바이러스 위치 저장
    for(int i = 0; i < N; i++)
    {
        for(int j = 0; j < M; j++)
        {
            cin >> arr[i][j];
            if     (arr[i][j] == 0) emptyList.emplace_back(i, j); // 빈칸 목록
            else if(arr[i][j] == 2) virusList.emplace_back(i, j); // 바이러스 목록
        }
    }

    // ═══════════════════════════════════════════════════════════════
    // ⭐ next_permutation 순열 메서드를 조합으로 응용하는 방법
    // ═══════════════════════════════════════════════════════════════
    vector<int> combi(emptyList.size(), 1);  // 모두 1로 초기화 (미선택 상태)
    fill_n(combi.begin(), 3, 0);             // 앞 3개만 0으로 변경 (선택 상태)

    // next_permutation으로 combi의 모든 순열 생성 (= 조합과 동일)
    do
    {
        // combi에서 0인 위치(선택된 위치)의 좌표를 picked에 저장
        vector<pair<int,int>> picked;
        for(int i = 0; i < (int)emptyList.size(); i++)
        {
            if(combi[i] == 0)  // 0 = 선택됨, 1 = 선택 안 됨
            {
                picked.push_back(emptyList[i]);
            }
        }

        // 선택된 3칸을 벽으로 세움
        for(auto pos : picked)
        {
            arr[pos.first][pos.second] = 1;
        }

        // 바이러스 확산 시뮬레이션 및 안전영역 계산
        ret = max(ret, Check());

        // 원복
        for(auto pos : picked)
        {
            arr[pos.first][pos.second] = 0;
        }

    } while(next_permutation(combi.begin(), combi.end()));

    cout << ret << "\n";
    return 0;
}
