#include <bits/stdc++.h>
using namespace std;

// n : 격자판 크기, a : 격자판 (0:빈칸, 1:벽)
// dp[y][x][d] : (y, x)에 파이프 끝이 d 방향으로 도달하는 경우의 수
//          d = 0 : 가로 (→)
//          d = 1 : 대각선 (↘)
//          d = 2 : 세로 (↓)
int n;
int a[24][24];
int dp[24][24][3];

// 파이프를 (y, x)에 d 방향으로 놓을 수 있는지 확인
// y : 행 좌표
// x : 열 좌표
// d : 파이프 방향 (0:가로, 1:대각선, 2:세로)
bool canPlace(int y, int x, int d)
{
    if(d == 0 || d == 2)
    {
        // 가로 또는 세로 : 해당 칸만 비어있으면 됨
        if(a[y][x] == 0)
            return true;
    }
    else if(d == 1)
    {
        // 대각선 : 해당 칸, 위쪽 칸, 왼쪽 칸 모두 비어있어야 함
        if(a[y][x] == 0 && a[y - 1][x] == 0 && a[y][x - 1] == 0)
            return true;
    }
    return false;
}

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    // 입력 처리
    cin >> n;
    for(int i = 1; i <= n; i++)
    {
        for(int j = 1; j <= n; j++)
        {
            cin >> a[i][j];
        }
    }

    // DP 초기값 : 파이프가 (1, 1)-(1, 2)에 가로로 놓여있음
    dp[1][2][0] = 1;

    // DP 계산 : 모든 위치에서 가능한 이동 탐색
    for(int i = 1; i <= n; i++)
    {
        for(int j = 1; j <= n; j++)
        {
            // ★ canPlace의 d자리에 방향을 넣어서, 1자리 or 3자리에 대해 유효한지 체크
            // ★ 다음 이동 자리(y,x)와 상태(d) += 현재 이동 자리(y,x)의 상태(d)에 저장된 값
            
            //    체크                     저장 자리               현재 자리 값
            // ★ 다름                     다름                    같음            
            
            // 현재 가로(0) 방향에서 이동 가능한 경우
            if(canPlace(i,     j + 1, 0)) dp[i    ][j + 1][0] += dp[i][j][0]; // 가로 → 가로
            if(canPlace(i + 1, j + 1, 1)) dp[i + 1][j + 1][1] += dp[i][j][0]; // 가로 → 대각선

            // 현재 대각선(1) 방향에서 이동 가능한 경우
            if(canPlace(i,     j + 1, 0)) dp[i    ][j + 1][0] += dp[i][j][1]; // 대각선 → 가로
            if(canPlace(i + 1, j + 1, 1)) dp[i + 1][j + 1][1] += dp[i][j][1]; // 대각선 → 대각선
            if(canPlace(i + 1, j,     2)) dp[i + 1][j    ][2] += dp[i][j][1]; // 대각선 → 세로

            // 현재 세로(2) 방향에서 이동 가능한 경우
            if(canPlace(i + 1, j,     2)) dp[i + 1][j    ][2] += dp[i][j][2]; // 세로 → 세로
            if(canPlace(i + 1, j + 1, 1)) dp[i + 1][j + 1][1] += dp[i][j][2]; // 세로 → 대각선
        }
    }

    // 결과 출력 : (n, n)에 도달하는 모든 경우의 수 합산
    cout << dp[n][n][0] + dp[n][n][1] + dp[n][n][2];

    return 0;
}
