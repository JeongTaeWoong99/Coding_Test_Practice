// ┌─────────────────────────────────────────────────────────────────┐
// │  SPFA (Shortest Path Faster Algorithm) — 일반 큐 버전           │
// ├─────────────────────────────────────────────────────────────────┤
// │  ✅ 정답은 나옴 / ❌ 최적 코드(Answer(정답))보다 느림           │
// ├─────────────────────────────────────────────────────────────────┤
// │  [정답 코드와의 차이]                                           
// │                                                                
// │  정답 코드 : priority_queue (min-heap) + continue 스킵         
// │  이 코드   : queue (FIFO)             + continue 제거          
// │                                                                
// │  [왜 일반 큐에선 continue를 쓰면 안 되는가?]                   
// │                                                                
// │  priority_queue는 비용이 작은 것을 먼저 꺼냄                 
// │  → 처음 꺼낸 순간 = 해당 도시의 최소 비용이 확정됨             
// │  → 이후 같은 도시가 다시 나오면 무조건 suboptimal              
// │  → continue 스킵이 항상 안전 ✅                                
// │                                                                
// │  일반 큐는 먼저 들어온 것을 먼저 꺼냄 (FIFO)                 
// │  → 처음 꺼낸 순간 = 최소 비용 보장 없음                        
// │  → 더 짧은 경로가 아직 큐 뒤쪽에 대기 중일 수 있음             
// │  → continue로 스킵하면 그 이후 간선 전파 기회를 잃어 오답 ❌   
// │                                                                
// │  [왜 continue 없이 정답이 나오는가?]                         
// │                                                                
// │  더 짧은 경로가 발견될 때마다 minCost를 갱신하고 큐에 재추가    
// │  → 결국 모든 최단 비용이 전파되어 수렴                          
// │  → Bellman-Ford의 큐 최적화 버전 (SPFA)                        
// │                                                                 
// │  [성능 비교]                                                    
// │  priority_queue : O(M log N)  — 이 문제 : 약 100,000×10 연산   
// │  일반 큐 (SPFA) : 평균 O(kM), 최악 O(NM) — 밀집 그래프에서 TLE 
// └─────────────────────────────────────────────────────────────────┘

#include <bits/stdc++.h>
using namespace std;

int n, m;                        // n : 도시 개수, m  : 버스(간선) 개수
int startCity, endCity;          // startCity  : 출발 도시,  endCity  : 도착 도시
int minCost[1001];               // minCost[i] : startCity 에서 i번 도시까지의 최소 비용
vector<pair<int,int>> adj[1001]; // adj[u] : u에서 출발하는 버스 목록 {비용, 도착 도시}

// SPFA — 일반 큐 버전
// priority_queue 대신 queue 사용, continue 스킵 제거
void SPFA()
{
    // 초기화 세팅
    fill(minCost, minCost + n + 1, INT_MAX); // 모든 도시의 최소 비용을 무한대로 초기화
    minCost[startCity] = 0;                  // 출발 도시는 비용 0

    // 일반 큐 : FIFO — 비용 순서 보장 없음
    queue<pair<int,int>> q;
    q.push({0, startCity});

    while (!q.empty())
    {
        int curCost = q.front().first;   // curCost : 현재 노드까지의 비용
        int curCity = q.front().second;  // curCity : 현재 처리 중인 도시 번호
        q.pop();

        // ❌ continue 스킵 없음
        // 일반 큐는 꺼낸 순간 최소 비용이 확정되지 않으므로 스킵 불가
        // → 비용이 suboptimal이어도 무조건 인접 도시 순회해야 함

        // curCity 에서 출발하는 모든 버스를 확인하여 인접 도시의 비용 갱신
        for (int i = 0; i < (int)adj[curCity].size(); ++i)
        {
            int edgeCost = adj[curCity][i].first;  // edgeCost  : 이 버스의 요금
            int nextCity = adj[curCity][i].second; // nextCity  : 이 버스의 도착 도시

            // totalCost : curCity 경유 시 nextCity 까지의 총 비용
            int totalCost = curCost + edgeCost;

            // 더 짧은 경로 발견 시 갱신 후 큐에 재추가
            // → 같은 도시가 큐에 여러 번 들어갈 수 있음 (비효율의 원인)
            if (totalCost < minCost[nextCity])
            {
                minCost[nextCity] = totalCost;
                q.push({totalCost, nextCity});
            }
        }
    }
}

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    // 도시 수, 버스 수 입력
    cin >> n >> m;

    // 버스 노선 입력 : 출발 도시, 도착 도시, 비용
    for (int i = 0; i < m; ++i)
    {
        int fromCity, toCity, weight;
        cin >> fromCity >> toCity >> weight;

        adj[fromCity].emplace_back(weight, toCity);
    }

    // 구간 입력 : 출발 도시 → 도착 도시
    cin >> startCity >> endCity;

    // SPFA 실행
    SPFA();

    // startCity → endCity 최소 비용 출력
    cout << minCost[endCity] << \n;

    return 0;
}
