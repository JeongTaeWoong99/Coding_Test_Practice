#include <bits/stdc++.h>
using namespace std;

typedef long long ll; // 좌표 곱셈 합산 시 오버플로우 방지

int  n;        // 꼭짓점 수
ll   x[10001]; // x 좌표 배열
ll   y[10001]; // y 좌표 배열

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    // 입력
    cin >> n;
    for (int i = 0; i < n; ++i)
    {
        cin >> x[i] >> y[i];
    }

    // 신발끈 공식으로 다각형 면적 계산
    ll cross = 0;
    for (int i = 0; i < n; ++i)
    {
        int next = (i + 1) % n; // 다음 꼭짓점 인덱스 (마지막은 0으로 순환)
        cross += x[i] * y[next] - x[next] * y[i];
    }

    // 절댓값 / 2 → 실제 면적 (소수점 첫째 자리까지 반올림)
    double area = abs(cross) / 2.0;

    cout << fixed << setprecision(1) << area << "\n";

    return 0;
}
