#include <bits/stdc++.h>
using namespace std;

// 전역 변수 선언
int n, a, b;              // n : 전깃줄 개수, a : A전봇대 위치, b : B전봇대 위치
int lis[504];             // lis : LIS를 저장할 배열 (B전봇대 위치의 증가 수열)
int len;                  // len : 현재 LIS의 길이 (교차 없이 남길 수 있는 최대 전깃줄 개수)
vector<pair<int, int>> v; // v : (A위치, B위치) 쌍을 저장하는 벡터

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(NULL);cout.tie(NULL);

    cin >> n;

    for(int i = 0; i < n; i++)
    {
        cin >> a >> b;
        v.emplace_back(a, b); // A와 B의 연결 위치를 쌍으로 저장
    }

    // A 전봇대 위치를 기준으로 오름차순 정렬
    // pair는 first 기준으로 자동 정렬됨
    sort(v.begin(), v.end());

    // B 전봇대 위치로 LIS 구하기 (이분 탐색 활용)
    for(int i = 0; i < v.size(); i++)
    {
        // lower_bound : lis 배열에서 v[i].second 이상인 첫 번째 위치를 찾음
        // pos : v[i].second가 들어갈 위치 (LIS를 유지하면서 삽입할 위치)
        // ★ v[i].second보다 큰 값이 없으면 -> lis + len 리턴 (마지막 위치)
        int pos = lower_bound(lis, lis + len, v[i].second) - lis;

        // ★ 마지막 위치(pos == len)에 삽입 -> 새로운 값 추가 -> LIS 길이 증가
        // ★ 기존 위치(pos < len)에 삽입 -> 값 교체 -> len 불변
        if(lis[pos] == 0) len++;

        // 해당 위치에 B 전봇대 위치 값을 삽입/갱신
        // - 기존 위치: 더 작은 값으로 교체 -> 미래에 더 긴 LIS 가능
        // - 마지막 위치: 새로운 값 추가 -> LIS 확장
        lis[pos] = v[i].second;
    }

    // 결과 출력 : 전체 전깃줄 - 남길 수 있는 최대 전깃줄 = 제거할 최소 전깃줄
    cout << n - len;

    return 0;
}
