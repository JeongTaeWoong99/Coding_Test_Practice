#include<bits/stdc++.h>
using namespace std;

const int dy[] = {-1, 1, 0, 0};
const int dx[] = {0, 0, 1, -1};

int  n, m, mx;          // n : 행 개수, m : 열 개수, mx : 최대 거리
char arr[54][54];       // 보물섬 지도
int  visited[54][54];   // 방문 및 거리 배열

// 각 육지(L)에서 BFS를 수행하여 최단 거리 계산
void BFS(int y, int x)
{
    // 모든 L 위치마다 BFS를 수행하므로 매번 초기화
    memset(visited, 0, sizeof(visited));

    visited[y][x] = 1;
    queue<pair<int, int>> bfsQ;
    bfsQ.emplace(y, x);

    while(!bfsQ.empty())
    {
        int nY, nX;
        tie(nY, nX) = bfsQ.front();
        bfsQ.pop();

        for(int i = 0; i < 4; i++)
        {
            int ny = nY + dy[i];
            int nx = nX + dx[i];

            // 범위 체크
            if(ny < 0 || ny >= n || nx < 0 || nx >= m) continue;

            // 방문 체크
            if(visited[ny][nx]) continue;

            // 바다(W)는 이동 불가
            if(arr[ny][nx] == 'W') continue;

            // 거리 갱신 (이전 칸 거리 + 1)
            visited[ny][nx] = visited[nY][nX] + 1;
            bfsQ.emplace(ny, nx);

            // 최대 거리 갱신
            mx = max(mx, visited[ny][nx]);
        }
    }
    return;
}

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    // 입력 처리
    cin >> n >> m;
    for(int i = 0; i < n; i++)
    {
        for(int j = 0; j < m; j++)
        {
            cin >> arr[i][j];
        }
    }

    // 모든 육지(L) 위치에서 BFS 수행
    for(int i = 0; i < n; i++)
    {
        for(int j = 0; j < m; j++)
        {
            if(arr[i][j] == 'L')
                BFS(i, j);
        }
    }

    // 결과 출력 (거리는 1부터 시작했으므로 -1)
    cout << mx - 1 << "\n";

    return 0;
}
