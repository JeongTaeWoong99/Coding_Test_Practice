#include <bits/stdc++.h>
using namespace std;

typedef long long ll; // 결과가 '2^63 - 1' 까지 나올 수 있으므로 ll 사용

int n;           // n    : 숫자 개수 (최대 100, int 충분)
int arr[104];    // arr  : 입력 숫자 배열 (값 범위 0~20, int 충분)
ll  dp[104][21]; // dp   : dp[idx][sum] = idx번째 인덱스에서 현재 합이 sum일 때의 경우의 수.

// idx : 현재 처리 중인 인덱스
// sum : 현재까지의 누적 합
ll DP(int idx, int sum)
{
    // 합이 0~20 범위를 벗어나면 유효하지 않은 경우
    if (sum < 0 || sum > 20)
    {
        return 0;
    }

    // ★ dp[idx][sum]에 대한 참조 : ret를 통한 읽기/쓰기가 dp 배열에 직접 반영됨
    ll &ret = dp[idx][sum];

    // 이미 계산된 값이면 바로 반환 (메모이제이션)
    if (ret)
    {
        return ret;
    }

    // 마지막 연산자 위치에 도달 : 합이 마지막 숫자와 같으면 유효한 경우
    if (idx == n - 2)
    {
        if (sum == arr[n - 1])
        {
            return 1;
        }

        return 0;
    }

    // 다음 숫자를 더하거나 빼는 두 가지 경우의 수 합산(바로 DP에 저장)
    ret += DP(idx + 1, sum + arr[idx + 1]);
    ret += DP(idx + 1, sum - arr[idx + 1]);

    return ret;
}

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    // 입력
    cin >> n;

    for (int i = 0; i < n; i++)
    {
        cin >> arr[i];
    }

    // 첫 번째 숫자를 초기 합으로 시작
    cout << DP(0, arr[0]) << "\n";

    return 0;
}
