#include <bits/stdc++.h>
using namespace std;

// 4방향 : {행, 열} 변화량 (상하좌우)
const int dY[4] = {-1,  1,  0,  0};
const int dX[4] = { 0,  0, -1,  1};

// 보관 후 하루가 지나면, 익은 토마토들의 인접한 곳에 있는 익지 않은 토마토들은 익은 토마토의 영향을 받아 익게 된다.
// 왼쪽, 오른쪽, 앞, 뒤 네 방향
// 정수 1  => 익은 토마토
// 정수 0  => 익지 않은 토마토
// 정수 -1 => 토마토가 들어있지 않은 칸

int m, n;
int arr[1001][1001];      // arr[행][열]
int visited[1001][1001];  // 방문 여부 체크 + 경과 일수 저장

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    cin >> m >> n;  // m : 가로 칸 수, n : 세로 칸 수

    // 입력
    for (int i = 0; i < n; i++)
    {
        for (int j = 0; j < m; j++)
        {
            cin >> arr[i][j];
        }
    }

    // BFS 시작 (큐) - 익은 토마토를 모두 큐에 넣기 (멀티 소스 BFS)
    queue<pair<int, int>> bfsQ;
    for (int i = 0; i < n; i++)
    {
        for (int j = 0; j < m; j++)
        {
            // 시작 큐에 넣기 (정수 1 => 익은 토마토)
            if (arr[i][j] == 1)
            {
                visited[i][j] = 1;    // 시작 지점 방문 체크
                bfsQ.emplace(i, j);   // 큐에 넣기
            }
        }
    }

    // 큐가 빌 때까지 진행
    while (!bfsQ.empty())
    {
        auto [ci, cj] = bfsQ.front();  // 큐 맨 앞 원소 가져오기
        bfsQ.pop();                    // 큐 맨 앞 원소 제거

        // 4방향 탐색 (상하좌우)
        for (int d = 0; d < 4; d++)
        {
            int ni = ci + dY[d];  // 새 행
            int nj = cj + dX[d];  // 새 열

            if (ni < 0 || nj < 0 || ni >= n || nj >= m) continue;  // 범위 나감
            if (arr[ni][nj] == -1)                       continue;  // 토마토 없는 칸
            if (visited[ni][nj])                         continue;  // 이미 방문

            visited[ni][nj] = visited[ci][cj] + 1;  // 이동 체크
            arr[ni][nj] = 1;                         // 익음 처리
            bfsQ.emplace(ni, nj);                    // 새로 넣기
        }
    }

    // 결과 계산
    int ans = 0;
    for (int i = 0; i < n; i++)
    {
        for (int j = 0; j < m; j++)
        {
            // 익지 못한 토마토 존재 => 불가 및 종료
            if (arr[i][j] == 0)
            {
                cout << -1 << "\n";
                return 0;
            }

            // 가장 큰 값 갱신
            ans = max(ans, visited[i][j]);
        }
    }

    // visited는 1부터 시작했으므로 -1
    cout << ans - 1 << "\n";

    return 0;
}
