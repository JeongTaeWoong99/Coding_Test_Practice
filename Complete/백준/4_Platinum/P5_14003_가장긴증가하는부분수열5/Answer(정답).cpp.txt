#include <bits/stdc++.h>
using namespace std;

const int          INF = 1e9 + 4;     // 무한대 값(10억 + 4)

int            n;                     // 수열의 크기
int            lis[1000004];          // LIS를 저장하는 배열
int            len;                   // 현재 LIS의 길이
int            num;                   // 입력받을 숫자
pair<int, int> ans[1000004];          // {LIS에서의 위치, 실제 값} 저장
stack<int>     stk;                   // 역추적 결과를 저장할 스택

int main()
{
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);

    // LIS 배열 초기화
    fill_n(lis, 1000004, INF);

    // 입력 처리
    cin >> n;

    // LIS 구하기 (이분 탐색)
    for(int i = 0; i < n; i++)
    {
        cin >> num;

        // lower_bound로 num이 들어갈 "인덱스" 찾기(주소 - 주소 = 인덱스)
        int pos = lower_bound(lis, lis + len, num) - lis;

        // 새로운 값이면 길이 증가
        if(lis[pos] == INF) len++;

        // LIS 배열 갱신
        lis[pos] = num;
        
        // 역추적을 위해 저장 : {LIS에서의 위치, 실제 값}
        ans[i].first  = pos;
        ans[i].second = num;
    }

    // LIS 길이 출력
    cout << len << "\n";

    // 역추적 : 뒤에서부터 LIS 복원
    for(int i = n - 1; i >= 0; i--)
    {
        if(ans[i].first == len - 1)
        {
            stk.push(ans[i].second);
            len--;
        }
    }

    // 스택에서 꺼내며 출력 (오름차순)
    while(!stk.empty())
    {
        cout << stk.top() << " ";
        stk.pop();
    }

    return 0;
}
