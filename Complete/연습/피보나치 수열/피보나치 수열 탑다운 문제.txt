// 탑다운이 압도적으로 유리한 경우: 희소 행렬 경로 탐색
// 문제 : (0,0)에서 (N-1, M-1)까지 가는 경로의 수 구하기
// 조건 : 오른쪽, 아래로만 이동 가능, 장애물(X)은 지나갈 수 없음

#include <bits/stdc++.h>
using namespace std;

int       N = 10, M = 10;
int       arr[11][11];     // 0 = 빈칸, 1 = 장애물
long long dp[11][11];      // -1로 초기화

// ============================================
// 탑다운 (Top-down) - 재귀 + 메모이제이션
// ============================================
long long TopDown(int y, int x)
{
    // 도착!
    if (y == N - 1 && x == M - 1) return 1;

    // 범위 밖 또는 장애물
    if (y >= N || x >= M || arr[y][x] == 1) return 0;

    // 이미 계산됨 (메모이제이션)
    long long &ret = dp[y][x];
    
    if (ret != -1) return ret;

    // 오른쪽 + 아래 경로의 합
    return ret = TopDown(y, x + 1) + TopDown(y + 1, x);
}

// ============================================
// 바텀업 (Bottom-up) - 반복문 + DP
// ============================================
long long BottomUp()
{
    long long dp2[11][11];
    memset(dp2, 0, sizeof(dp2));

    dp2[0][0] = 1; // 시작점

    // 모든 칸을 순회 (필요 없는 칸도 체크!)
    for (int y = 0; y < N; y++)
    {
        for (int x = 0; x < M; x++)
        {
            if (arr[y][x] == 1) continue; // 장애물

            // 위에서 아래로 오는 경로(범위 안 && 빈칸)
            // 방문 자리 += 이전 값 누적
            if (y > 0 && arr[y - 1][x] == 0)
                dp2[y][x] += dp2[y - 1][x];

            // 왼쪽에서 오른쪽으로 오는 경로(범위 안 && 빈칸)
            // 방문 자리 += 이전 값 누적
            if (x > 0 && arr[y][x - 1] == 0)
                dp2[y][x] += dp2[y][x - 1];
        }
    }

    return dp2[N - 1][M - 1];
}

int main()
{
    ios_base::sync_with_stdio(false); cin.tie(0); cout.tie(0);

    // 지도 초기화 (0 = 빈칸, 1 = 장애물)
    memset(arr, 0, sizeof(arr));

    // 예시 1: 장애물이 거의 없는 경우
    cout << "=== 예시 1: 장애물 없음 ===" << "\n";
    memset(dp, -1, sizeof(dp));
    cout << "탑다운 결과 : " << TopDown(0, 0) << "\n";
    cout << "바텀업 결과 : " << BottomUp() << "\n";
    cout << "→ 계산량 비슷함, 바텀업이 약간 더 빠름\n\n";

    // 예시 2: 장애물이 매우 많은 경우 (희소 행렬)
    cout << "=== 예시 2: 장애물 매우 많음 (희소 행렬) ===" << "\n";

    // 대각선만 비어있는 경로 (나머지는 모두 장애물)
    for (int i = 0; i < N; i++)
        for (int j = 0; j < M; j++)
            arr[i][j] = 1; // 모두 장애물로

    // 대각선 경로만 열기
    for (int i = 0; i < N; i++)
        arr[i][i] = 0;

    memset(dp, -1, sizeof(dp));
    cout << "탑다운 결과: " << TopDown(0, 0) << "\n";
    cout << "바텀업 결과: " << BottomUp() << "\n";
    cout << "\n";

    cout << "=== 성능 분석 ===" << "\n";
    cout << "탑다운: 실제 경로만 탐색 (~20개 칸)\n";
    cout << "바텀업: 모든 칸 순회 (100개 칸)\n";
    cout << "→ 계산량 5배 차이! 탑다운이 더 효율적\n\n";

    cout << "=== 핵심 차이점 ===" << "\n";
    cout << "탑다운:\n";
    cout << "  - 재귀로 필요한 경로만 탐색\n";
    cout << "  - 장애물 만나면 즉시 백트래킹 (return 0)\n";
    cout << "  - 도달 불가능한 칸은 아예 방문 안 함!\n\n";

    cout << "바텀업:\n";
    cout << "  - for문으로 모든 칸을 순회\n";
    cout << "  - 장애물도 체크해야 함 (if문으로)\n";
    cout << "  - 도달 불가능한 칸도 모두 확인함\n\n";

    cout << "결론: 희소 행렬처럼 실제 계산이 필요한 칸이 적을 때\n";
    cout << "      탑다운이 압도적으로 유리!\n";

    return 0;
}
