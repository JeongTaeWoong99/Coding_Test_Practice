#include <bits/stdc++.h>
using namespace std;

// <R C J A> // 첫번째 알파벳, 비동의(1~3) 시
// 1 - 3점
// 2 - 2점
// 3 - 1점

// <중앙>
// 4 - 점수 없음

// <T F M N> // 두번째 알파벳, 동의(5~7) 시
// 5 - 1점
// 6 - 2점
// 7 - 3점
string solution(vector<string> survey, vector<int> choices)
{
    unordered_map<char, int> score; // 각 지표(R/T/C/F/J/M/A/N)별 점수 맵핑 (정렬 불필요 → 해시맵)

    // survey.size() = choices.size() 숫자만큼 체크 진행
    for (int i = 0; i < survey.size(); ++i)
    {
        // ※ survey의 알파벳 순서는 바뀔 수 있음. => RT or TR
        char curFirstAlpa  = survey[i][0]; // 첫번째 자리 알파벳 + 비동의(1~3) 시 가리키는 지표
        char curSecondAlpa = survey[i][1]; // 두번째 자리 알파벳 + 동의(5~7)   시 가리키는 지표
        int  curChoice     = choices[i];

        // 4(모르겠음)는 제외
        if      (curChoice < 4) score[curFirstAlpa]  += 4         - curChoice; // 1→3, 2→2, 3→1점
        else if (curChoice > 4) score[curSecondAlpa] += curChoice - 4;         // 5→1, 6→2, 7→3점
    }

    string answer = "";
    string axis   = "RTCFJMAN"; // 두 글자씩 한 쌍 (앞 글자가 알파벳 우선)

    for (int i = 0; i < 8; i += 2)
    {
        char a = axis[i];     // R C J A (모두 알파벳 상 앞에 옴)
        char b = axis[i + 1]; // T F M N

        // 동점이면 알파벳 빠른 a 선택
        answer += (score[a] >= score[b]) ? a : b;
    }

    return answer;
}

int main(int argc, char* argv[])
{
    ios_base::sync_with_stdio(0),cin.tie(0),cout.tie(0);

    cout << solution({"AN", "CF", "MJ", "RT", "NA"}, {5, 3, 2, 7, 5}) << '\n'; // TCMA
    cout << solution({"TR", "RT", "TR"},             {7, 1, 3})       << '\n'; // RCJA

    return 0;
}
