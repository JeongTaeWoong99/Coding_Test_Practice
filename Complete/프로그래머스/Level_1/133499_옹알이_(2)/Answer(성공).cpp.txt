#include <bits/stdc++.h>
using namespace std;

vector<string> possibleSound = {"aya", "ye", "woo", "ma"}; // 발음 가능한 4개

// 하나의 단어가 발음 가능한지 검사 (같은 발음 연속 금지)
bool CanSpeak(string word)
{
    int    idx  = 0;  // 현재 검사 위치
    string prev = ""; // 직전에 사용한 발음

    // 모든 문자 비교
    while (idx < (int)word.size())
    {
        bool matched = false; // 이번 위치에서 매칭 성공 여부

        // 가능한 모든 문자와 비교해보기
        for (string& curS : possibleSound)
        {
            // 직전과 같은 발음은 연속 금지
            if (curS == prev)
            {
                continue;
            }

            // 현재 위치에서 발음s 와 일치하는지 확인(idx부터 s.size()까지 word와 s 비교)
            if (word.compare(idx, curS.size(), curS) == 0)
            {
                idx    += (int)curS.size(); // 비교 시작 위치 변경
                prev    = curS;             // 이전 발음 저장
                matched = true;             // 매치 되는 발음 발견

                break;
            }
        }

        // 어떤 발음과도 매칭 안 되면 발음 불가
        if (!matched)
        {
            return false;
        }
    }

    return true;
}

int solution(vector<string> babbling)
{
    int answer = 0; // 발음 가능한 단어 수

    for (string& word : babbling)
    {
        if (CanSpeak(word))
        {
            answer++;
        }
    }

    return answer;
}
