#include <bits/stdc++.h>
using namespace std;

// ingredient를 스택에 쌓으며 맨 위 4개가 빵-야채-고기-빵(1-2-3-1)이면 햄버거 완성
int solution(vector<int> ingredient)
{
    int         answer = 0; // 완성한 햄버거 수
    vector<int> stk;        // 재료를 쌓는 스택

    for (int cur : ingredient)
    {
        stk.emplace_back(cur);

        int totalSize = stk.size(); // 현재 쌓인 재료 수

        // 맨 위 4개가 1-2-3-1이면 햄버거 1개 완성
        if (totalSize >= 4 && stk[totalSize - 4] == 1 && 
                              stk[totalSize - 3] == 2 && 
                              stk[totalSize - 2] == 3 && 
                              stk[totalSize - 1] == 1)
        {
            stk.resize(totalSize - 4); // 위에서부터 4개를 잘라내 햄버거 재료 제거

            answer++;
        }
    }

    return answer;
}

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    cout << solution({2, 1, 1, 2, 3, 1, 2, 3, 1}) << "\n"; // 기대값 2
    cout << solution({1, 3, 2, 1, 2, 1, 3, 1, 2}) << "\n"; // 기대값 0

    return 0;
}
