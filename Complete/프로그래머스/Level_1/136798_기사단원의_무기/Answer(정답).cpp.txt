#include <bits/stdc++.h>
using namespace std;

// number번 기사까지 각 기사 번호의 약수 개수를 구해 무게 합산
// 약수 개수 > limit 이면 power로 대체
int solution(int number, int limit, int power)
{
    int totalWeight = 0;

    for (int i = 1; i <= number; i++)
    {
        int divisorCnt = 0;

        // sqrt(i)까지만 탐색해 약수 쌍으로 개수 계산
        for (int j = 1; j * j <= i; j++)
        {
            if (i % j == 0)
            {
                divisorCnt++;

                // j ≠ i/j : 서로 다른 약수 쌍 존재 → 둘 다 추가 (예: i=12, j=2 → {2, 6})
                // j == i/j : 완전제곱수의 제곱근 → 하나만 추가 (예: i=9, j=3 → {3})
                if (j != i / j)
                {
                    divisorCnt++;
                }
            }
        }

        // limit 초과 시 power로 대체
        if (divisorCnt > limit)
        {
            divisorCnt = power;
        }

        totalWeight += divisorCnt;
    }

    return totalWeight;
}
