#include <bits/stdc++.h>
using namespace std;

int a[26];

// 각 알파벳의 마지막 등장 위치를 배열로 추적, 현재 위치와의 차이 계산
vector<int> solution(string s)
{
    vector<int> answer;

    fill_n(a, 26, -1);

    for (int i = 0; i < s.length(); i++)
    {
        // 처음 나온 경우
        if (a[s[i] - 'a'] == -1)
        {
            answer.push_back(-1); // 처음 발견은 -1 넣기

            a[s[i] - 'a'] = i;    // 발견 자리값으로 변경
        }
        // 이후 등장
        else
        {
            answer.push_back(i - (a[s[i] - 'a'])); // 차이값 = 현재 자리값 - 이전에 기록된 자리값

            a[s[i] - 'a'] = i;                     // 발견 자리값으로 변경
        }
    }

    return answer;
}

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    vector<int> ans = solution("banana");

    for (const auto& num : ans)
    {
        cout << num << " ";
    }

    return 0;
}
