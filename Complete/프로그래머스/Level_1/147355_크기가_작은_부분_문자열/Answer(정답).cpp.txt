#include <bits/stdc++.h>
using namespace std;

int solution(string t, string p)
{
    int answer  = 0;
    int tLength = t.length();
    int pLength = p.length();

    for (int i = 0; i <= tLength - pLength; ++i)
    {
        // 문자열 직접 비교(둘은 같은 n자릿수)
        if (t.substr(i, pLength) <= p)
            answer++;
    }

    return answer;
}
