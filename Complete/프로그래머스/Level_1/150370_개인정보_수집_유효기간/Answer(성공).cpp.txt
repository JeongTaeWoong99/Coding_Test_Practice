#include <bits/stdc++.h>
using namespace std;

// 날짜 문자열을 총 일수로 변환 (모든 달 = 28일 기준)
// month - 1 : month는 1~12이므로 0-indexed(0~11)로 변환
int ToTotalDays(const string& date)
{
    int year  = stoi(date.substr(0, 4));
    int month = stoi(date.substr(5, 2));
    int day   = stoi(date.substr(8, 2));

    return (year * 12 + month - 1) * 28 + day;
}

vector<int> solution(string today, vector<string> terms, vector<string> privacies)
{
    vector<int> answer;

    // 약관 종류(char) → 유효기간(월) 매핑
    map<char, int> termMap;

    for (string& t : terms)
    {
        termMap[t[0]] = stoi(t.substr(2)); // t[0] : 약관 종류, t.substr(2) : 유효기간(월)
    }

    int todayDays = ToTotalDays(today);

    for (int i = 0; i < (int)privacies.size(); i++)
    {
        int privacyDays = ToTotalDays(privacies[i].substr(0, 10)); // 수집일 (앞 10자리)
        int term        = termMap[privacies[i][11]];               // 약관 종류는 index 11의 char

        // 만료일이 오늘 이하면 파기 대상
        if (privacyDays + term * 28 <= todayDays)
        {
            answer.emplace_back(i + 1); // 1-indexed
        }
    }

    return answer;
}
