#include <bits/stdc++.h>
using namespace std;

// 이웃한 칸 — board[r][c]와 같은 색인 상하좌우 이웃 칸 수 반환
int solution(vector<vector<string>> board, int r, int c)
{
    int    answer = 0;
    int    n      = board.size();
    string color  = board[r][c];

    int dr[] = {-1, 1,  0, 0};
    int dc[] = { 0, 0, -1, 1};

    for (int i = 0; i < 4; i++)
    {
        int nr = r + dr[i];
        int nc = c + dc[i];

        if (nr < 0 || nr >= n || nc < 0 || nc >= n) continue;
        
        if (board[nr][nc] == color) answer++;
    }

    return answer;
}

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    // 테스트 케이스 1 : 예상 출력 2
    cout << solution({{"blue","red","orange","red"},{"red","red","blue","orange"},{"blue","orange","red","red"},{"orange","orange","red","blue"}}, 1, 1) << "\n";

    // 테스트 케이스 2 : 예상 출력 1
    cout << solution({{"yellow","green","blue"},{"blue","green","yellow"},{"green","blue","yellow"}}, 1, 1) << "\n";

    return 0;
}
