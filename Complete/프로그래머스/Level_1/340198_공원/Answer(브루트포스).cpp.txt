#include <bits/stdc++.h>
using namespace std;

int solution(vector<int> mats, vector<vector<string>> park)
{
    int totalRows = park.size();    // 세로
    int totalCols = park[0].size(); // 가로(원소 갯수가 곧 가로 길이)

    // 큰 mat부터 확인(가능한 제일 큰 값을 찾는거니 바로 리턴하면 됨)
    sort(mats.rbegin(), mats.rend());

    // 모든 mats 값 확인
    for (int curNum : mats)
    {
        // 세로 시작 위치
        for (int curRow = 0; curRow <= totalRows - curNum; curRow++)
        {
            // 가로 시작 위치
            for (int curCol = 0; curCol <= totalCols - curNum; curCol++)
            {
                bool isPossible = true;

                // 2차원 배열 범위 확인
                for (int i = curRow; i < curRow + curNum && isPossible; i++)
                {
                    for (int j = curCol; j < curCol + curNum && isPossible; j++)
                    {
                        if (park[i][j] != "-1")
                            isPossible = false;
                    }
                }

                // 가장 큰 mat 찾으면 즉시 리턴
                if (isPossible)
                {
                    return curNum;
                }
            }
        }
    }

    // 모두 불가능
    return -1;
}
