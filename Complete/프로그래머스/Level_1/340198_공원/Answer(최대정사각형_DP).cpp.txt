#include <bits/stdc++.h>
using namespace std;

int solution(vector<int> mats, vector<vector<string>> park)
{
    int totalRows = park.size();    // 세로
    int totalCols = park[0].size(); // 가로(원소 갯수가 곧 가로 길이)

    // dp[i][j]: (i, j)를 우하단으로 하는 최대 빈 정사각형 크기
    vector<vector<int>> dp(totalRows, vector<int>(totalCols, 0));
    int maxSide = 0;

    for (int i = 0; i < totalRows; i++)
    {
        for (int j = 0; j < totalCols; j++)
        {
            if (park[i][j] != "-1")
                continue;

            // 첫 행/열은 최대 1x1
            if (i == 0 || j == 0)
                dp[i][j] = 1;
            else
                dp[i][j] = min({ dp[i - 1][j], dp[i][j - 1], dp[i - 1][j - 1] }) + 1;

            maxSide = max(maxSide, dp[i][j]);
        }
    }

    // 내림차순 정렬 후, 놓을 수 있는 가장 큰 mat 리턴
    sort(mats.rbegin(), mats.rend());
    
    for (int curNum : mats)
    {
        if (curNum <= maxSide)
            return curNum;
    }
    
    // 모두 불가능
    return -1;
}
