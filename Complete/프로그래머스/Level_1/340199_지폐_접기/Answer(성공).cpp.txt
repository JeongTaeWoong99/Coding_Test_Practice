#include <bits/stdc++.h>
using namespace std;

// wallet 크기에 맞게 bill을 접는 최소 횟수 반환
int solution(vector<int> wallet, vector<int> bill)
{
    int answer = 0;

    while (true)
    {
        // 지폐가 지갑에 들어가는지 확인 (회전 고려)
        if ((bill[0] <= wallet[0] && bill[1] <= wallet[1]) ||
            (bill[0] <= wallet[1] && bill[1] <= wallet[0]))
        {
            break;
        }

        // 긴 변을 반으로 접기
        if (bill[0] >= bill[1])
        {
            bill[0] /= 2;
        }
        else
        {
            bill[1] /= 2;
        }

        answer++;
    }

    return answer;
}
