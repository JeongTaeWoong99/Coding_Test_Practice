#include <bits/stdc++.h>
using namespace std;

// "mm:ss" 형식의 시간을 초 단위 정수로 변환
int ToSec(const string& t)
{
    return stoi(t.substr(0, 2)) * 60 + stoi(t.substr(3, 2));
}

// 초 단위 정수를 "mm:ss" 형식의 문자열로 변환
string ToStr(int sec)
{
    string mm = (sec / 60 < 10 ? "0" : "") + to_string(sec / 60);
    string ss = (sec % 60 < 10 ? "0" : "") + to_string(sec % 60);

    return mm + ":" + ss;
}

string solution(string video_len, string pos, string op_start, string op_end, vector<string> commands)
{
    int videoLen = ToSec(video_len);
    int curPos   = ToSec(pos);
    int opStart  = ToSec(op_start);
    int opEnd    = ToSec(op_end);

    // 시작 위치가 오프닝 구간이면 op_end로 이동
    if (curPos >= opStart && curPos <= opEnd)
    {
        curPos = opEnd;
    }

    for (const string& cmd : commands)
    {
        if (cmd == "prev")
        {
            curPos -= 10;
            curPos  = max(curPos, 0);
        }
        // "next"
        else
        {
            curPos += 10;
            curPos  = min(curPos, videoLen);
        }

        // 커맨드 처리 후 오프닝 구간이면 op_end로 이동
        if (curPos >= opStart && curPos <= opEnd)
        {
            curPos = opEnd;
        }
    }

    return ToStr(curPos);
}

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    // 테스트 케이스 1: "13:00" 예상
    cout << solution("34:33", "13:00", "00:55", "02:55", {"next", "prev"}) << "\n";

    // 테스트 케이스 2: "06:55" 예상
    cout << solution("10:55", "00:05", "00:15", "06:55", {"prev", "next", "next"}) << "\n";

    // 테스트 케이스 3: "04:17" 예상
    cout << solution("07:22", "04:05", "00:15", "04:07", {"next"}) << "\n";

    return 0;
}
