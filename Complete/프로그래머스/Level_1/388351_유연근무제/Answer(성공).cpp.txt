#include <bits/stdc++.h>
using namespace std;

// HHMM 형식을 분 단위로 변환
int ToMinutes(int hhmm)
{
    return (hhmm / 100) * 60 + (hhmm % 100);
}

// 이벤트 기간(7일) 동안 모든 평일에 정상 출근한 직원 수 반환
// schedules[i] : 직원 i의 출근 희망 시각 (HHMM)
// timelogs[i]  : 직원 i의 7일간 실제 출근 시각 (HHMM)
// startday     : 이벤트 시작 요일 (1=월, 7=일)
int solution(vector<int> schedules, vector<vector<int>> timelogs, int startday)
{
    int answer = 0;
    int n      = (int)schedules.size();

    for (int i = 0; i < n; i++)
    {
        int  limitTime = ToMinutes(schedules[i]) + 10; // 출근 마감 시각 (분 단위)
        bool onTime    = true;                         // 제시간에 모두 들어왔는지

        // 일주일 확인
        for (int j = 0; j < 7; j++)
        {
            // j번째 날의 요일 계산 (1=월 2=화 ~ 6=토 7=일)
            int dayOfWeek = (startday - 1 + j) % 7 + 1;

            // 토요일(6), 일요일(7) 스킵
            if (dayOfWeek == 6 || dayOfWeek == 7)
            {
                continue;
            }

            // 마감 시각 초과 시 조기 탈락
            if (ToMinutes(timelogs[i][j]) > limitTime)
            {
                onTime = false;
                break;
            }
        }

        // 모든 평일 제시간에 출근함
        if (onTime)
        {
            answer++;
        }
    }

    return answer;
}

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    // 테스트 케이스
    cout << solution({700, 800, 1100}, {{710, 2359, 1050, 700, 650, 631, 659}, {800, 801, 805, 800, 759, 810, 809}, {1105, 1001, 1002, 600, 1059, 1001, 1100}}, 5) << "\n"; // 3
    cout << solution({730, 855, 700, 720}, {{710, 700, 650, 735, 700, 931, 912}, {908, 901, 805, 815, 800, 831, 835}, {705, 701, 702, 705, 710, 710, 711}, {707, 731, 859, 913, 934, 931, 905}}, 1) << "\n"; // 2

    return 0;
}
