#include <bits/stdc++.h>
using namespace std;

// 각 명함을 회전시켜 긴 변은 한 축(A), 짧은 변은 다른 축(B)으로 몰아넣고
// 두 축의 최댓값을 곱해 모든 명함을 담는 최소 지갑 넓이를 구한다
int solution(vector<vector<int>> sizes)
{
    int maxA = 0; // 긴 변들 중 최댓값 (지갑의 한 변)
    int maxB = 0; // 짧은 변들 중 최댓값 (지갑의 다른 변)

    for (auto& card : sizes)
    {
        int longSide  = max(card[0], card[1]); // 명함의 긴 변
        int shortSide = min(card[0], card[1]); // 명함의 짧은 변

        maxA = max(maxA, longSide);
        maxB = max(maxB, shortSide);
    }

    return maxA * maxB;
}
