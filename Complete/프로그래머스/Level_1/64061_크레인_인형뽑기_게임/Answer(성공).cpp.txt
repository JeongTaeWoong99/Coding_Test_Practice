#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> board =
{
    { 0, 0, 0, 0, 0 },
    { 0, 0, 1, 0, 3 },
    { 0, 2, 5, 0, 1 },
    { 4, 2, 4, 4, 2 },
    { 3, 5, 1, 3, 1 }
};
vector<int> moves = { 1, 5, 3, 5, 1, 2, 1, 4 };

// board의 각 열을 스택으로 만들어 크레인으로 뽑은 인형을 바구니에 쌓고, 바구니 맨 위와 같으면 터뜨림
int solution(vector<vector<int>> board, vector<int> moves)
{
    int n      = board.size(); // 격자 크기
    int answer = 0;            // 터져서 사라진 인형 수

    vector<vector<int>> columns(n); // columns[j] = j번 열의 인형들 (back = 맨 위 인형)
    vector<int>         basket;     // 뽑은 인형을 쌓는 바구니 스택

    // 아래 행부터 위로 올라가며 0이 아닌 인형만 쌓기 → back()이 맨 위 인형
    for (int j = 0; j < n; j++)
    {
        for (int i = n - 1; i >= 0; i--)
        {
            if (board[i][j] != 0)
            {
                columns[j].emplace_back(board[i][j]);
            }
        }
    }

    // 크레인 작동 순서대로 처리
    for (int move : moves)
    {
        int col = move - 1; // moves는 1번 열부터 시작하므로 인덱스로 변환

        if (columns[col].empty()) // 뽑을 인형이 없는 열은 스킵
        {
            continue;
        }

        int doll = columns[col].back(); // 맨 위 인형 뽑기
        columns[col].pop_back();

        // 바구니 맨 위와 같은 인형이면 둘 다 터짐
        if (!basket.empty() && basket.back() == doll)
        {
            basket.pop_back();

            answer += 2;
        }
        else
        {
            basket.emplace_back(doll);
        }
    }

    return answer;
}

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    cout << solution(board, moves) << "\n"; // 기대값 4

    return 0;
}
