#include <bits/stdc++.h>
using namespace std;

// 문자열 양쪽 끝의 마침표(.) 제거
string RemoveEdgeDot(string s)
{
    // 앞쪽 마침표 제거
    while (!s.empty() && s.front() == '.')
    {
        s.erase(s.begin());
    }

    // 뒤쪽 마침표 제거
    while (!s.empty() && s.back() == '.')
    {
        s.pop_back();
    }

    return s;
}

string solution(string new_id)
{
    string answer = "";

    // 규칙에 맞지 않을 때, 유사한 아이디 추천
    // 1. 아이디의 길이는 3자 이상 15자 이하
    // 2. 알파벳 소문자, 숫자, 빼기(-), 밑줄(_), 마침표(.) 문자만 사용
    // ※ 단, 마침표(.)는 처음과 끝에 사용할 수 없으며 또한 연속으로 사용할 수 없습니다.

    // 1단계 : 모든 대문자를 소문자로 치환
    for (char& c : new_id)
    {
        c = tolower(c);
    }

    // 2단계 : 허용 문자(소문자, 숫자, -, _, .)를 제외한 나머지 제거
    for (char c : new_id)
    {
        if (islower(c) || isdigit(c) || c == '-' || c == '_' || c == '.')
        {
            answer += c;
        }
    }

    // 3단계 : 연속된 마침표(.)를 하나로 치환
    string temp = "";

    for (char c : answer)
    {
        if (c == '.' && !temp.empty() && temp.back() == '.') // 직전 문자도 '.' 이면 스킵
        {
            continue;
        }

        temp += c;
    }

    answer = temp;

    // 4단계 : 처음과 끝의 마침표(.) 제거
    answer = RemoveEdgeDot(answer);

    // 5단계 : 빈 문자열이면 "a" 대입
    if (answer.empty())
    {
        answer = "a";
    }

    // 6단계 : 16자 이상이면 15자로 자르고, 끝의 마침표(.) 제거
    if (answer.size() >= 16)
    {
        answer = answer.substr(0, 15);
        answer = RemoveEdgeDot(answer);
    }

    // 7단계 : 2자 이하이면 마지막 문자를 3자가 될 때까지 반복
    while (answer.size() <= 2)
    {
        answer += answer.back();
    }

    return answer;
}
