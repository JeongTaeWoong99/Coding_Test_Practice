#include <bits/stdc++.h>
using namespace std;

// 맞춘 개수를 로또 순위로 변환 (6개 중 n개 일치 → 순위)
int GetRank(int matchCnt)
{
    // 2개 미만은 모두 6등(순위 없음), 그 외는 7 - 맞춘개수
    if (matchCnt < 2)
    {
        return 6;
    }

    return 7 - matchCnt;
}

// lottos : 내 로또 번호(0 = 알아볼 수 없는 번호), win_nums : 당첨 번호
vector<int> solution(vector<int> lottos, vector<int> win_nums)
{
    int zeroCnt  = 0; // 알아볼 수 없는 번호(0)의 개수
    int matchCnt = 0; // 알아볼 수 있는 번호 중 당첨 번호와 일치하는 개수

    // 당첨 번호 빠른 조회를 위한 집합
    set<int> winSet(win_nums.begin(), win_nums.end());

    for (int num : lottos)
    {
        // 0이면 알아볼 수 없는 번호
        if (num == 0)
        {
            zeroCnt++;
        }
        // 당첨 번호에 포함되면 일치 카운트
        else if (winSet.count(num))
        {
            matchCnt++;
        }
    }

    // 최고 순위 : 0을 모두 맞춘 것으로 가정, 최저 순위 : 0을 모두 틀린 것으로 가정
    int highRank = GetRank(matchCnt + zeroCnt);
    int lowRank  = GetRank(matchCnt);

    return {highRank, lowRank};
}
