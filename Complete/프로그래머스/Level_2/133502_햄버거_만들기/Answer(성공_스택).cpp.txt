#include <bits/stdc++.h>

using namespace std;

// ingredient를 스택에 쌓으며 맨 위 4개가 빵-야채-고기-빵(1-2-3-1)이면 햄버거 완성
int solution(vector<int> ingredient)
{
    int         answer = 0; // 완성한 햄버거 수
    vector<int> stk;        // 재료를 쌓는 스택

    for (int cur : ingredient)
    {
        stk.emplace_back(cur);

        int top = stk.size(); // 현재 쌓인 재료 수

        // 맨 위 4개가 1-2-3-1이면 햄버거 1개 완성
        if (top >= 4 && stk[top - 4] == 1 && stk[top - 3] == 2 && stk[top - 2] == 3 && stk[top - 1] == 1)
        {
            stk.resize(top - 4); // 사용한 재료 4개 제거

            answer++;
        }
    }

    return answer;
}
