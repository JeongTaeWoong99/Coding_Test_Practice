#include <bits/stdc++.h>
using namespace std;

// [비밀 코드 해독]
// 1 ~ num 중 5개로 이루어진 '비밀 코드'가 숨겨져 있음.
// 각 시도(tries[i])는 5개의 숫자이고, hint[i] = (비밀 코드와 겹치는 개수).
// → 모든 힌트를 동시에 만족하는 비밀 코드 후보가 몇 개인지 세는 문제.

int                 num;    // 숫자 범위 (1 ~ num 중 5개 선택)
vector<vector<int>> tries;  // 각 시도 (5개씩)
vector<int>         hint;   // 각 시도의 힌트 (비밀 코드와 겹친 개수)
int                 answer; // 조건을 만족하는 비밀 코드 후보 개수

// 뽑은 5개 조합(code)이 모든 시도의 힌트와 정확히 일치하는지 검사
bool Check(vector<int>& code)
{
    set<int> picked(code.begin(), code.end()); // 뽑힌 숫자 집합

    for (int i = 0; i < (int)tries.size(); i++)
    {
        int overlapCnt = 0;

        // i번째 시도에서 뽑힌 숫자와 겹치는 개수
        for (int x : tries[i])
        {
            if (picked.count(x))
            {
                overlapCnt++;
            }
        }

        // 힌트와 정확히 일치하지 않으면 탈락
        if (overlapCnt != hint[i])
        {
            return false;
        }
    }

    return true;
}

// start부터 숫자를 골라 5개짜리 조합을 모두 생성 (완전탐색)
void DFS(int start, vector<int>& code)
{
    // 5개를 다 뽑았으면 힌트 검사
    if (code.size() == 5)
    {
        if (Check(code))
        {
            answer++;
        }

        return;
    }

    // start ~ num 중 하나를 골라 다음 자리로
    for (int i = start; i <= num; i++)
    {
        code.emplace_back(i);

        DFS(i + 1, code);

        code.pop_back();
    }
}

int solution(int n, vector<vector<int>> q, vector<int> ans)
{
    num    = n;   // 숫자 범위 (1 ~ num 중 5개 선택)
    tries  = q;   // 각 시도 (5개씩)
    hint   = ans; // 각 시도의 힌트 (비밀 코드와 겹친 개수)
    answer = 0;   // 조건을 만족하는 비밀 코드 후보 개수

    vector<int> code; // 현재 만들고 있는 비밀 코드 후보

    DFS(1, code);

    return answer;
}
