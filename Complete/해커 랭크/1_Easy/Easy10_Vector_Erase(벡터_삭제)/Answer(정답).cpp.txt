#include <bits/stdc++.h>
using namespace std;

int         n, p, a, b; // n : 벡터 크기, p : 단일 삭제 위치, a, b : 범위 삭제 위치
vector<int> v;           // 입력 벡터

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    // 넣기
    cin >> n;

    v.resize(n);
    for (int i = 0; i < n; i++)
    {
        cin >> v[i];
    }

    // 단일 원소 삭제 (1-indexed)
    cin >> p;

    v.erase(v.begin() + p - 1);

    // 범위 삭제 (1-indexed, [a, b) — b는 exclusive)
    cin >> a >> b;

    v.erase(v.begin() + a - 1, v.begin() + b - 1);

    // 결과 출력
    cout << v.size() << "\n";
    for (int x : v)
    {
        cout << x << " ";
    }

    return 0;
}
