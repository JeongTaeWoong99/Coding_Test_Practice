#include <bits/stdc++.h>
using namespace std;

int         n;
vector<int> v;

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    cin >> n;

    v.resize(n);

    for (int i = 0; i < n; i++)
    {
        cin >> v[i];
    }

    sort(v.begin(), v.end()); // 오름차순 정렬

    for (int x : v)
    {
        cout << x << " ";
    }

    return 0;
}
