#include <bits/stdc++.h>
using namespace std;

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    int T;
    cin >> T;

    while (T--)
    {
        double A, B, C;
        cin >> A >> B >> C;

        // ── A 출력 ──────────────────────────────────────────────────────
        // 소수점 버리고(truncate) → 16진수, 0x prefix, 소문자
        // 예) 100.345 → (long long)100 → 0x64
        cout << hex                    // 출력 진법을 16진수로 변경
             << showbase               // 0x prefix 자동 추가
             << nouppercase            // 소문자 유지 (0x, a~f)
             << (long long)A << "\n";  // truncate 후 출력 (hex는 정수형에만 동작)

        // ── B 출력 ──────────────────────────────────────────────────────
        // 2자리 소수, +/- 부호, 전체 15자 너비, 빈 자리는 _ 채움 (우측 정렬)
        // 예) 2006.008 → "+2006.01" (8자) → "_______+2006.01" (15자)
        cout << dec                 // hex → dec 복원 (필수)
             << setw(15)            // 총 15자 너비 (한 번만 적용)
             << setfill('_')        // 빈 자리 _ 채움
             << showpos             // 양수에 + 부호
             << fixed               // 고정 소수점
             << setprecision(2)     // 소수 2자리
             << B << "\n";

        // ── C 출력 ──────────────────────────────────────────────────────
        // 과학적 표기법, 소수 9자리, 대문자 E
        // 예) 2331.41592653498 → 2.331415927E+03
        cout << noshowpos           // B에서 켠 + 부호 제거 (필수)
             << uppercase           // E, X 대문자
             << scientific          // 과학적 표기법
             << setprecision(9)     // 소수 9자리
             << C << "\n";
    }

    return 0;
}
