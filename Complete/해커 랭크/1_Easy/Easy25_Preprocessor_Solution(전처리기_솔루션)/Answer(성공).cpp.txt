#include <bits/stdc++.h>
using namespace std;

#define INF      10000000
#define NEG_INF  -10000000
#define MAX(a,b) ((a) > (b) ? (a) : (b))
#define MIN(a,b) ((a) < (b) ? (a) : (b))

int n, x;

int main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

    cin >> n;

    int mn = INF;     // 최솟값 (INF로 초기화)
    int mx = NEG_INF; // 최댓값 (NEG_INF로 초기화)

    for (int i = 0; i < n; i++)
    {
        cin >> x;

        mn = MIN(mn, x);
        mx = MAX(mx, x);
    }

    cout << "Result = " << mx - mn << endl;

    return 0;
}
